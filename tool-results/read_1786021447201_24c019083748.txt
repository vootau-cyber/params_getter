  4345→// Helper: get empty row for a given section
  4346→// =============================================================================
  4347→
  4348→export function getEmptyRow(sectionKey: string): Record<string, unknown> {
  4349→  const section = SCHEMA_SECTIONS.find((s) => s.key === sectionKey);
  4350→  if (!section) return {};
  4351→
  4352→  const row: Record<string, unknown> = {};
  4353→  for (const field of section.fields) {
  4354→    // Skip virtual fields — they are UI-only and not saved
  4355→    if (field.virtual) continue;
  4356→
  4357→    if (field.type === "object" && field.nestedFields) {
  4358→      if (Array.isArray(field.defaultValue)) {
  4359→        row[field.key] = field.defaultValue;
  4360→      } else {
  4361→        row[field.key] = {};
  4362→      }
  4363→    } else if (field.type === "array") {
  4364→      row[field.key] = [];
  4365→    } else if (field.type === "boolean") {
  4366→      row[field.key] = field.defaultValue ?? false;
  4367→    } else if (field.type === "number") {
  4368→      row[field.key] = field.defaultValue ?? null;
  4369→    } else if (field.type === "date") {
  4370→      row[field.key] = field.defaultValue ?? null;
  4371→    } else if (field.type === "ref") {
  4372→      row[field.key] = field.defaultValue ?? null;
  4373→    } else {
  4374→      row[field.key] = field.defaultValue ?? "";
  4375→    }
  4376→  }
  4377→  return row;
  4378→}

  4380→/**
  4381→ * Returns the set of virtual field keys for each section.
  4382→ * Used to strip UI-only fields before saving/exporting.
  4383→ */
  4384→export function getVirtualFieldKeys(): Record<string, Set<string>> {
  4385→  const map: Record<string, Set<string>> = {};
  4386→  for (const section of SCHEMA_SECTIONS) {
  4387→    const virtualKeys = new Set<string>();
  4388→    for (const field of section.fields) {
  4389→      if (field.virtual) virtualKeys.add(field.key);
  4390→    }
  4391→    if (virtualKeys.size > 0) {
  4392→      map[section.key] = virtualKeys;
  4393→    }
  4394→  }
  4395→  return map;
  4396→}

  4398→/**
  4399→ * Returns refAutoFill mappings for virtual fields, keyed by section.
  4400→ * { sectionKey: { virtualFieldKey: { targetKey: sourceKey } } }
  4401→ */
  4402→export function getRefAutoFillMappings(): Record<string, Record<string, Record<string, string>>> {
  4403→  const map: Record<string, Record<string, Record<string, string>>> = {};
  4404→  for (const section of SCHEMA_SECTIONS) {
  4405→    const sectionMappings: Record<string, Record<string, string>> = {};
  4406→    for (const field of section.fields) {
  4407→      if (field.virtual && field.refAutoFill) {
  4408→        sectionMappings[field.key] = field.refAutoFill;
  4409→      }
  4410→    }
  4411→    if (Object.keys(sectionMappings).length > 0) {
  4412→      map[section.key] = sectionMappings;
  4413→    }
  4414→  }
  4415→  return map;
