    63→function TestStatusIcon({ result }: { result: ConnectionTestResult | undefined }) {
    64→  if (!result) return null;
    65→  if (result.ok) {
    66→    return <CheckCircle className="size-3.5 text-emerald-500 shrink-0" />;
    67→  }
    68→  if (result.error && result.error === 'Тестирование…') {
    69→    return <Loader2 className="size-3.5 animate-spin text-muted-foreground shrink-0" />;
    70→  }
    71→  return <XCircle className="size-3.5 text-destructive shrink-0" />;

   361→// =============================================================================
   362→// SQL Table Data Grid (inline)
   363→// =============================================================================
   364→
   365→function SQLTableDataGrid({
   366→  tableData,
   367→  loading,
   368→  onLoadPage,
   369→  tableName,
   370→  onBack,
   371→  onImport,
   372→}: {
   373→  tableData: { columns: string[]; rows: Record<string, unknown>[]; total_count: number } | null;
   374→  loading: boolean;
   375→  onLoadPage: (offset: number) => void;
   376→  tableName: string;
   377→  onBack: () => void;
   378→  onImport: () => void;
   379→}) {
   380→  const [offset, setOffset] = useState(0);
   381→  const pageSize = 50;
   382→
   383→  const handlePrev = () => {
   384→    const newOffset = Math.max(0, offset - pageSize);
   385→    setOffset(newOffset);
   386→    onLoadPage(newOffset);
   387→  };
   388→
   389→  const handleNext = () => {
   390→    if (!tableData) return;
   391→    const newOffset = offset + pageSize;
   392→    if (newOffset < tableData.total_count) {
   393→      setOffset(newOffset);
   394→      onLoadPage(newOffset);
   395→    }
   396→  };
   397→
   398→  return (
   399→    <div className="space-y-3">
   400→      {/* Breadcrumb */}
   401→      <div className="flex items-center gap-2 text-sm">
   402→        <Button variant="ghost" size="sm" className="h-6 px-2 text-xs" onClick={onBack}>
   403→          <ChevronRight className="size-3 rotate-180 mr-1" />
   404→          Таблицы
   405→        </Button>
   406→        <ChevronRight className="size-3 text-muted-foreground" />
   407→        <span className="font-medium">{tableName}</span>
   408→        <Badge variant="outline" className="text-xs ml-2">
   409→          {tableData?.total_count ?? 0} строк
   410→        </Badge>
   411→      </div>
   412→
   413→      {/* Table */}
   414→      {loading ? (
   415→        <div className="flex items-center justify-center py-12">
   416→          <Loader2 className="size-6 animate-spin text-muted-foreground" />
   417→        </div>
   418→      ) : tableData && tableData.columns.length > 0 ? (
   419→        <div className="border rounded-md overflow-hidden">
   420→          <div className="max-h-96 overflow-y-auto">
   421→            <table className="w-full text-xs">
   422→              <thead className="bg-muted/50 sticky top-0">
   423→                <tr>
   424→                  <th className="text-left px-3 py-2 font-medium text-muted-foreground w-10">
   425→                    #
   426→                  </th>
   427→                  {tableData.columns.map((col) => (
   428→                    <th
   429→                      key={col}
   430→                      className="text-left px-3 py-2 font-medium text-muted-foreground whitespace-nowrap max-w-[200px] truncate"
   431→                    >
   432→                      {col}
   433→                    </th>
   434→                  ))}
   435→                </tr>
   436→              </thead>
   437→              <tbody>
   438→                {tableData.rows.map((row, idx) => (
   439→                  <tr
   440→                    key={idx}
   441→                    className="border-t border-border/50 hover:bg-muted/30 transition-colors"
   442→                  >
   443→                    <td className="px-3 py-1.5 text-muted-foreground">{offset + idx + 1}</td>
   444→                    {tableData.columns.map((col) => (
   445→                      <td key={col} className="px-3 py-1.5 whitespace-nowrap max-w-[200px] truncate">
   446→                        {row[col] !== null && row[col] !== undefined
   447→                          ? String(row[col])
   448→                          : '—'}
   449→                      </td>
   450→                    ))}
   451→                  </tr>
   452→                ))}
   453→              </tbody>
   454→            </table>
   455→          </div>
   456→        </div>
   457→      ) : (
   458→        <div className="flex items-center justify-center py-12 text-muted-foreground text-sm">
   459→          Нет данных
   460→        </div>
   461→      )}
   462→
   463→      {/* Pagination */}
   464→      <div className="flex items-center justify-between">
   465→        <Button
   466→          variant="outline"
   467→          size="sm"
   468→          className="h-7 text-xs"
   469→          disabled={offset === 0}
   470→          onClick={handlePrev}
   471→        >
   472→          ← Назад
   473→        </Button>
   474→        <span className="text-xs text-muted-foreground">
   475→          {offset + 1}–{Math.min(offset + pageSize, tableData?.total_count ?? 0)} из{' '}
   476→          {tableData?.total_count ?? 0}
   477→        </span>
   478→        <Button
   479→          variant="outline"
   480→          size="sm"
   481→          className="h-7 text-xs"
   482→          disabled={!tableData || offset + pageSize >= tableData.total_count}
   483→          onClick={handleNext}
   484→        >
   485→          Далее →
   486→        </Button>
   487→      </div>
   488→
   489→      {/* Import button */}
   490→      <Separator />
   491→      <div className="flex items-center justify-end">
   492→        <Button variant="outline" size="sm" onClick={onImport}>
   493→          <Upload className="size-3.5 mr-1.5" />
   494→          Импортировать в раздел
   495→        </Button>
   496→      </div>
   497→    </div>
   498→  );
   499→}

   502→// SQL Import Form
   503→// =============================================================================
   504→
   505→function SQLImportForm({
   506→  tableName,
   507→  columns,
   508→  onImport,
   509→  onCancel,
   510→  importing,
   511→}: {
   512→  tableName: string;
   513→  columns: string[];
   514→  onImport: (sectionKey: string, mapping: Record<string, string>) => void;
   515→  onCancel: () => void;
   516→  importing: boolean;
   517→}) {
   518→  // Compute initial auto-map and best section key synchronously
   519→  const [initialState] = useState(() => {
   520→    const initialMapping: Record<string, string> = {};
   521→    const sectionCounts: Record<string, number> = {};
   522→    for (const col of columns) {
   523→      const allFields = SCHEMA_SECTIONS.flatMap((s) => s.fields);
   524→      const match = allFields.find(
   525→        (f) => f.key.toLowerCase() === col.toLowerCase(),
   526→      );
   527→      if (match) {
   528→        initialMapping[col] = match.key;
   529→        for (const section of SCHEMA_SECTIONS) {
   530→          if (section.fields.some((f) => f.key === match.key)) {
   531→            sectionCounts[section.key] = (sectionCounts[section.key] || 0) + 1;
   532→          }
   533→        }
   534→      }
   535→    }
   536→    const best = Object.entries(sectionCounts).sort((a, b) => b[1] - a[1])[0];
   537→    return { mapping: initialMapping, sectionKey: best ? best[0] : '' };
   538→  });
   539→
   540→  const [sectionKey, setSectionKey] = useState(initialState.sectionKey);
   541→  const [mapping, setMapping] = useState<Record<string, string>>(initialState.mapping);
   542→
   543→  const selectedSection = useMemo(
   544→    () => SCHEMA_SECTIONS.find((s) => s.key === sectionKey),
   545→    [sectionKey],
   546→  );
   547→
   548→  const editableFields = useMemo(
   549→    () => (selectedSection?.fields || []).filter((f) => !f.virtual && !f.readOnly),
   550→    [selectedSection],
   551→  );
   552→
   553→  const handleImport = () => {
   554→    if (!sectionKey) {
   555→      toast.error('Выберите целевой раздел');
   556→      return;
   557→    }
   558→    // Filter out unmapped columns
   559→    const cleanMapping: Record<string, string> = {};
   560→    for (const [col, field] of Object.entries(mapping)) {
   561→      if (field) cleanMapping[col] = field;
   562→    }
   563→    if (Object.keys(cleanMapping).length === 0) {
   564→      toast.error('Настройте хотя бы одно соответствие полей');
   565→      return;
   566→    }
   567→    onImport(sectionKey, cleanMapping);
   568→  };
   569→
   570→  return (
   571→    <div className="space-y-4">
   572→      <div className="flex items-center gap-2 mb-2">
   573→        <Upload className="size-4 text-primary" />
   574→        <h3 className="text-sm font-semibold">
   575→          Импорт: {tableName}
   576→        </h3>
   577→      </div>
   578→
   579→      {/* Target section */}
   580→      <div className="space-y-1.5">
   581→        <Label className="text-xs">Целевой раздел</Label>
   582→        <Select value={sectionKey} onValueChange={setSectionKey}>
   583→          <SelectTrigger className="h-8 text-sm">
   584→            <SelectValue placeholder="Выберите раздел…" />
   585→          </SelectTrigger>
   586→          <SelectContent>
   587→            {SCHEMA_SECTIONS.map((s) => (
   588→              <SelectItem key={s.key} value={s.key}>
   589→                {s.label}
   590→              </SelectItem>
   591→            ))}
   592→          </SelectContent>
   593→        </Select>
   594→      </div>
   595→
   596→      {/* Field mapping */}
   597→      {selectedSection && (
   598→        <div className="space-y-1.5">
   599→          <Label className="text-xs">Соответствие полей</Label>
   600→          <div className="border rounded-md max-h-64 overflow-y-auto">
   601→            <table className="w-full text-xs">
   602→              <thead className="bg-muted/50 sticky top-0">
   603→                <tr>
   604→                  <th className="text-left px-3 py-1.5 font-medium text-muted-foreground">
   605→                    Столбец SQL
   606→                  </th>
   607→                  <th className="text-left px-3 py-1.5 font-medium text-muted-foreground">
   608→                    <ArrowRight className="size-3 inline-block mr-1" />
   609→                    Поле раздела
   610→                  </th>
   611→                </tr>
   612→              </thead>
   613→              <tbody>
   614→                {columns.map((col) => (
   615→                  <tr key={col} className="border-t border-border/50">
   616→                    <td className="px-3 py-1.5 font-mono">{col}</td>
   617→                    <td className="px-3 py-1.5">
   618→                      <Select
   619→                        value={mapping[col] || ''}
   620→                        onValueChange={(v) =>
   621→                          setMapping((prev) => ({ ...prev, [col]: v }))
   622→                        }
   623→                      >
   624→                        <SelectTrigger className="h-7 text-xs">
   625→                          <SelectValue placeholder="— не сопоставлено —" />
   626→                        </SelectTrigger>
   627→                        <SelectContent>
   628→                          <SelectItem value="">— не сопоставлено —</SelectItem>
   629→                          {editableFields.map((f) => (
   630→                            <SelectItem key={f.key} value={f.key}>
   631→                              {f.label} ({f.key})
   632→                            </SelectItem>
