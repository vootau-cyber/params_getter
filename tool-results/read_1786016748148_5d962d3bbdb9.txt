  4282→// Helper: get empty row for a given section
  4283→// =============================================================================
  4284→
  4285→export function getEmptyRow(sectionKey: string): Record<string, unknown> {
  4286→  const section = SCHEMA_SECTIONS.find((s) => s.key === sectionKey);
  4287→  if (!section) return {};
  4288→
  4289→  const row: Record<string, unknown> = {};
  4290→  for (const field of section.fields) {
  4291→    if (field.type === "object" && field.nestedFields) {
  4292→      if (Array.isArray(field.defaultValue)) {
  4293→        row[field.key] = field.defaultValue;
  4294→      } else {
  4295→        row[field.key] = {};
  4296→      }
  4297→    } else if (field.type === "array") {
  4298→      row[field.key] = [];
  4299→    } else if (field.type === "boolean") {
  4300→      row[field.key] = field.defaultValue ?? false;
  4301→    } else if (field.type === "number") {
  4302→      row[field.key] = field.defaultValue ?? null;
  4303→    } else if (field.type === "date") {
  4304→      row[field.key] = field.defaultValue ?? null;
  4305→    } else if (field.type === "ref") {
  4306→      row[field.key] = field.defaultValue ?? null;
  4307→    } else {
  4308→      row[field.key] = field.defaultValue ?? "";
  4309→    }
  4310→  }
  4311→  return row;
  4312→}
