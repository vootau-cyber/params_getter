    55→  function validateField(key, value, type) {
    56→    if (value === null || value === undefined || String(value).trim() === '') {
    57→      return 'empty';
    58→    }
    59→
    60→    /* Additional type-based validation */
    61→    if (type === 'number') {
    62→      var numCheck = Number(value);
    63→      if (!isFinite(numCheck)) return 'invalid';
    64→    }
    65→    if (type === 'date') {
    66→      var dateStr = String(value).trim();
    67→      if (!/^\d{4}-\d{2}-\d{2}$/.test(dateStr)) return 'invalid';
    68→      var dateObj = new Date(dateStr);
    69→      if (isNaN(dateObj.getTime())) return 'invalid';
    70→    }
    71→
    72→    for (var i = 0; i < VALIDATION_RULES.length; i++) {
    73→      var rule = VALIDATION_RULES[i];
    74→      if (!rule.keyPattern.test(key)) continue;
    75→      var str = String(value).trim();
    76→      if (rule.isLat) {
    77→        var n = parseFloat(str);
    78→        if (isNaN(n) || n < -90 || n > 90) return 'invalid';
    79→        continue;
    80→      }
    81→      if (rule.isLon) {
    82→        var n2 = parseFloat(str);
    83→        if (isNaN(n2) || n2 < -180 || n2 > 180) return 'invalid';
    84→        continue;
    85→      }
    86→      if (rule.minLen !== undefined) {
    87→        if (str.length < rule.minLen) return 'invalid';
    88→        continue;
    89→      }
    90→      if (rule.test && !rule.test.test(str)) return 'invalid';
    91→    }
    92→    return 'ok';
    93→  }

   192→  function esc(str) {
   193→    if (str === null || str === undefined) return '';
   194→    return String(str)
   195→      .replace(/&/g, '&amp;')
   196→      .replace(/</g, '&lt;')
   197→      .replace(/>/g, '&gt;')
   198→      .replace(/"/g, '&quot;');
   199→  }

   201→  function deepClone(obj) {
   202→    return JSON.parse(JSON.stringify(obj));
   203→  }

   205→  function getSectionDef(sectionKey) {
   206→    if (!APP.schema || !APP.schema.sections) return null;
   207→    for (var i = 0; i < APP.schema.sections.length; i++) {
   208→      if (APP.schema.sections[i].key === sectionKey) return APP.schema.sections[i];
   209→    }
   210→    return null;
   211→  }
   212→
   213→  function getFieldDef(sectionKey, fieldKey) {
   214→    var sec = getSectionDef(sectionKey);
   215→    if (!sec || !sec.fields) return null;
   216→    for (var i = 0; i < sec.fields.length; i++) {
   217→      if (sec.fields[i].key === fieldKey) return sec.fields[i];
   218→    }
   219→    return null;
   220→  }
   221→
   222→  function isVirtualField(sectionKey, fieldKey) {
   223→    var fd = getFieldDef(sectionKey, fieldKey);
   224→    return fd && fd.virtual === true;
   225→  }

   227→  function stripVirtuals(data) {
   228→    var cleaned = deepClone(data);
   229→    if (!APP.schema || !APP.schema.sections) return cleaned;
   230→    for (var s = 0; s < APP.schema.sections.length; s++) {
   231→      var secKey = APP.schema.sections[s].key;
   232→      if (!cleaned[secKey]) continue;
   233→      var virtualKeys = [];
   234→      for (var f = 0; f < APP.schema.sections[s].fields.length; f++) {
   235→        if (APP.schema.sections[s].fields[f].virtual) {
   236→          virtualKeys.push(APP.schema.sections[s].fields[f].key);
   237→        }
   238→      }
   239→      for (var r = 0; r < cleaned[secKey].length; r++) {
   240→        for (var vk = 0; vk < virtualKeys.length; vk++) {
   241→          delete cleaned[secKey][r][virtualKeys[vk]];
   242→        }
   243→      }
   244→    }
   245→    return cleaned;
   246→  }

   253→  }

   512→  function renderField(sectionKey, rowIndex, fieldDef, value) {
   513→    var key = fieldDef.key;
   514→    var label = fieldDef.label || key;
   515→    var type = fieldDef.type || 'text';
   516→    var readOnly = fieldDef.readOnly === true;
   517→    var autoFilled = fieldDef.autoFilled === true;
   518→    var isVirtual = fieldDef.virtual === true;
   519→    var options = fieldDef.options || [];
   520→    var nestedFields = fieldDef.nestedFields || [];
   521→    var refSection = fieldDef.refSection;
   522→    var refLabelField = fieldDef.refLabelField;
   523→
   524→    var status = validateField(key, value, type);
   525→    var statusClass = '';
   526→    var tooltipHtml = '';
   527→
   528→    if (status === 'empty') {
   529→      statusClass = ' field-status-empty';
   530→    } else if (status === 'invalid') {
   531→      statusClass = ' invalid-field';
   532→      var hint = getFieldHint(key);
   533→      tooltipHtml = '<span class="invalid-tooltip">Поле ' + esc(label) + ' — некорректный формат ввода. Корректный формат: ' + esc(hint) + '</span>';
   534→    }
   535→
   536→    var autoBadge = '';
   537→    if (autoFilled || isVirtual) {
   538→      autoBadge = ' <span class="badge-auto">авто</span>';
   539→    }
   540→
   541→    var dataAttrs = ' data-section="' + esc(sectionKey) + '" data-row="' + rowIndex + '" data-field="' + esc(key) + '"';
   542→
   543→    var labelHtml = '<label class="field-label" style="display:flex;align-items:center;gap:6px;">' +
   544→      (status === 'empty' && !readOnly ? '<span style="width:6px;height:6px;border-radius:50%;background:var(--destructive);flex-shrink:0;"></span>' : '') +
   545→      esc(label) +
   546→      autoBadge +
   547→      tooltipHtml +
   548→    '</label>';
   549→
   550→    var inputHtml = '';
   551→
   552→    if (type === 'object') {
   553→      var nestedData = Array.isArray(value) ? value : [];
   554→      inputHtml = renderNestedTable(sectionKey, rowIndex, key, label, nestedFields, nestedData, readOnly);
   555→      return '<div class="field-wrapper" style="margin-bottom:16px;">' + inputHtml + '</div>';
   556→    }
   557→
   558→    var disabledAttr = readOnly ? ' disabled' : '';
   559→    var roClass = readOnly ? ' style="opacity:0.6;background:var(--muted);"' : '';
   560→
   561→    if (type === 'boolean') {
   562→      var checked = value === true || value === 'true' || value === 1 ? ' checked' : '';
   563→      inputHtml = '<div class="field-checkbox-wrapper">' +
   564→        '<input type="checkbox" class="field-input"' + dataAttrs + checked + disabledAttr + ' />' +
   565→      '</div>';
   566→    } else if (type === 'select' && options.length > 0) {
   567→      inputHtml = '<select class="field-select' + statusClass + '"' + dataAttrs + disabledAttr + roClass + '>';
   568→      inputHtml += '<option value=""></option>';
   569→      for (var o = 0; o < options.length; o++) {
   570→        var opt = options[o];
   571→        var optVal = typeof opt === 'object' ? opt.value : opt;
   572→        var optLabel = typeof opt === 'object' ? opt.label : opt;
   573→        var sel = (String(value) === String(optVal)) ? ' selected' : '';
   574→        inputHtml += '<option value="' + esc(optVal) + '"' + sel + '>' + esc(optLabel) + '</option>';
   575→      }
   576→      inputHtml += '</select>';
   577→    } else if (type === 'ref' || type === 'virtual') {
   578→      inputHtml = renderRefSelect(sectionKey, rowIndex, key, fieldDef, value, statusClass, disabledAttr, roClass, isVirtual);
   579→    } else if (type === 'textarea') {
   580→      inputHtml = '<textarea class="field-textarea' + statusClass + '"' + dataAttrs + disabledAttr + roClass + ' rows="3">' + esc(value) + '</textarea>';
   581→    } else if (type === 'number') {
   582→      inputHtml = '<input type="number" class="field-number' + statusClass + '"' + dataAttrs + ' value="' + esc(value) + '"' + disabledAttr + roClass + ' />';
   583→    } else if (type === 'date') {
   584→      var dateVal = value ? String(value).substring(0, 10) : '';
   585→      inputHtml = '<input type="date" class="field-date' + statusClass + '"' + dataAttrs + ' value="' + esc(dateVal) + '"' + disabledAttr + roClass + ' />';
   586→    } else if (type === 'array') {
   587→      var arrStr = Array.isArray(value) ? value.join(', ') : (value || '');
   588→      inputHtml = '<input type="text" class="field-input' + statusClass + '"' + dataAttrs + ' value="' + esc(arrStr) + '"' + disabledAttr + roClass + ' placeholder="Значения через запятую" />';
   589→    } else {
   590→      inputHtml = '<input type="text" class="field-input' + statusClass + '"' + dataAttrs + ' value="' + esc(value) + '"' + disabledAttr + roClass + ' />';
   591→    }
   592→
   593→    return '<div class="field-wrapper" style="margin-bottom:16px;">' + labelHtml + inputHtml + '</div>';
   594→  }

   695→  function pluralRu(n, one, few, many) {
   696→    var abs = Math.abs(n) % 100;
   697→    var lastDigit = abs % 10;
   698→    if (abs > 10 && abs < 20) return many;
   699→    if (lastDigit > 1 && lastDigit < 5) return few;
   700→    if (lastDigit === 1) return one;
   701→    return many;
   702→  }

  1149→  function handleFieldInput(sectionKey, rowIndex, fieldKey, target, e) {
  1150→    if (!APP.data[sectionKey] || !APP.data[sectionKey][rowIndex]) return;
  1151→    var fd = getFieldDef(sectionKey, fieldKey);
  1152→    if (!fd) return;
  1153→    if (fd.readOnly === true) return; /* don't update read-only fields */
  1154→    var type = fd.type || 'text';
  1155→
  1156→    if (type === 'boolean') {
  1157→      APP.data[sectionKey][rowIndex][fieldKey] = target.checked;
  1158→    } else if (type === 'number') {
  1159→      var numVal = target.value === '' ? null : parseFloat(target.value);
  1160→      APP.data[sectionKey][rowIndex][fieldKey] = numVal;
  1161→    } else if (type === 'array') {
  1162→      var strVal = target.value.trim();
  1163→      APP.data[sectionKey][rowIndex][fieldKey] = strVal ? strVal.split(',').map(function (s) { return s.trim(); }).filter(function (s) { return s !== ''; }) : [];
  1164→    } else if (type === 'ref' || type === 'virtual') {
  1165→      /* ref/virtual handled in change, but also update on input for text fallback */
  1166→      APP.data[sectionKey][rowIndex][fieldKey] = target.value;
  1167→    } else {
  1168→      APP.data[sectionKey][rowIndex][fieldKey] = target.value;
  1169→    }
  1170→
  1171→    checkDirty();
  1172→  }
  1173→
  1174→  function handleFieldChange(sectionKey, rowIndex, fieldKey, target, e) {
  1175→    if (!APP.data[sectionKey] || !APP.data[sectionKey][rowIndex]) return;
  1176→    var fd = getFieldDef(sectionKey, fieldKey);
  1177→    if (!fd) return;
  1178→    if (fd.readOnly === true) return; /* don't update read-only fields */
  1179→    var type = fd.type || 'text';
  1180→
  1181→    if (type === 'boolean') {
  1182→      APP.data[sectionKey][rowIndex][fieldKey] = target.checked;
  1183→      checkDirty();
  1184→      return;
  1185→    }
  1186→
  1187→    if (type === 'ref') {
  1188→      APP.data[sectionKey][rowIndex][fieldKey] = target.value;
  1189→      /* Check infra ref auto-fill */
  1190→      if (INFRA_REF_MAP[fieldKey]) {
  1191→        handleInfraRefChange(sectionKey, rowIndex, fieldKey, target.value);
  1192→      }
  1193→      checkDirty();
  1194→      return;
  1195→    }
  1196→
  1197→    if (type === 'virtual') {
  1198→      APP.data[sectionKey][rowIndex][fieldKey] = target.value;
  1199→      handleVirtualRefChange(sectionKey, rowIndex, fieldKey, target.value);
  1200→      checkDirty();
  1201→      renderApp();
  1202→      return;
  1203→    }
  1204→
  1205→    if (type === 'select') {
  1206→      APP.data[sectionKey][rowIndex][fieldKey] = target.value;
  1207→      checkDirty();
  1208→      return;
  1209→    }
  1210→
  1211→    /* For number type on change (spin buttons, etc.) */
  1212→    if (type === 'number') {
  1213→      var numVal2 = target.value === '' ? null : parseFloat(target.value);
  1214→      APP.data[sectionKey][rowIndex][fieldKey] = numVal2;
  1215→      checkDirty();
  1216→      return;
  1217→    }
  1218→  }

  1366→  /*
  1367→   * updateFieldValidationStatus - re-renders a single field's validation
  1368→   * indicators (red dot, invalid border, tooltip) without a full re-render.
  1369→   * This provides instant visual feedback as the user types.
  1370→   */
  1371→  function updateFieldValidationStatus(sectionKey, rowIndex, fieldKey) {
  1372→    var fd = getFieldDef(sectionKey, fieldKey);
  1373→    if (!fd) return;
  1374→    var value = (APP.data[sectionKey] || [])[rowIndex] || {};
  1375→    value = value[fieldKey];
  1376→    var type = fd.type || 'text';
  1377→
  1378→    var status = validateField(fieldKey, value, type);
  1379→    var label = fd.label || fieldKey;
  1380→    var hint = getFieldHint(fieldKey);
  1381→
  1382→    /* Find the field wrapper by data attributes */
  1383→    var selector = '.field-wrapper [data-section="' + sectionKey + '"][data-row="' + rowIndex + '"][data-field="' + fieldKey + '"]';
  1384→    var inputEl = document.querySelector(selector);
  1385→    if (!inputEl) return;
  1386→
  1387→    /* Update input border classes */
  1388→    if (status === 'invalid') {
  1389→      inputEl.classList.add('invalid-field');
  1390→    } else {
  1391→      inputEl.classList.remove('invalid-field');
  1392→    }
  1393→
  1394→    /* Update the wrapper's label area */
  1395→    var wrapper = inputEl.closest('.field-wrapper');
  1396→    if (!wrapper) return;
  1397→    var labelEl = wrapper.querySelector('.field-label');
  1398→    if (labelEl) {
  1399→      /* Remove old dot and tooltip */
  1400→      var oldDot = labelEl.querySelector('.val-dot');
  1401→      if (oldDot) oldDot.remove();
  1402→      var oldTooltip = labelEl.querySelector('.invalid-tooltip');
  1403→      if (oldTooltip) oldTooltip.remove();
  1404→
  1405→      if (status === 'empty' && fd.readOnly !== true) {
  1406→        var dot = document.createElement('span');
  1407→        dot.className = 'val-dot';
  1408→        dot.style.cssText = 'width:6px;height:6px;border-radius:50%;background:var(--destructive);flex-shrink:0;display:inline-block;';
  1409→        labelEl.insertBefore(dot, labelEl.firstChild);
  1410→      }
  1411→
  1412→      if (status === 'invalid') {
  1413→        var tip = document.createElement('span');
  1414→        tip.className = 'invalid-tooltip';
  1415→        tip.textContent = 'Поле ' + label + ' — некорректный формат ввода. Корректный формат: ' + hint;
  1416→        labelEl.appendChild(tip);
  1417→      }
  1418→    }
  1419→  }

  1422→     24. SECTION SUMMARY STATS
  1423→     ------------------------------------------------------------------ */
  1424→
  1425→  function getSectionStats(sectionKey) {
  1426→    var rows = APP.data[sectionKey] || [];
  1427→    if (rows.length === 0) return '';
  1428→    var filled = 0;
  1429→    var total = 0;
  1430→    var secDef = getSectionDef(sectionKey);
  1431→    if (!secDef || !secDef.fields) return '' + rows.length;
  1432→    for (var r = 0; r < rows.length; r++) {
  1433→      for (var f = 0; f < secDef.fields.length; f++) {
  1434→        var fk = secDef.fields[f].key;
  1435→        if (secDef.fields[f].virtual) continue;
  1436→        total++;
  1437→        var v = rows[r][fk];
  1438→        if (v !== null && v !== undefined && String(v).trim() !== '' && !(Array.isArray(v) && v.length === 0)) {
  1439→          filled++;
  1440→        }
  1441→      }
  1442→    }
  1443→    if (total === 0) return '';
  1444→    var pct = Math.round((filled / total) * 100);
  1445→    return rows.length + ' · ' + pct + '%';
  1446→  }
