  4230→// Helper: get empty row for a given section
  4231→// =============================================================================
  4232→
  4233→export function getEmptyRow(sectionKey: string): Record<string, unknown> {
  4234→  const section = SCHEMA_SECTIONS.find((s) => s.key === sectionKey);
  4235→  if (!section) return {};
  4236→
  4237→  const row: Record<string, unknown> = {};
  4238→  for (const field of section.fields) {
  4239→    if (field.type === "object" && field.nestedFields) {
  4240→      if (Array.isArray(field.defaultValue)) {
  4241→        row[field.key] = field.defaultValue;
  4242→      } else {
  4243→        row[field.key] = {};
  4244→      }
  4245→    } else if (field.type === "array") {
  4246→      row[field.key] = [];
  4247→    } else if (field.type === "boolean") {
  4248→      row[field.key] = field.defaultValue ?? false;
  4249→    } else if (field.type === "number") {
  4250→      row[field.key] = field.defaultValue ?? null;
  4251→    } else if (field.type === "date") {
  4252→      row[field.key] = field.defaultValue ?? null;
  4253→    } else if (field.type === "ref") {
  4254→      row[field.key] = field.defaultValue ?? null;
  4255→    } else {
  4256→      row[field.key] = field.defaultValue ?? "";
  4257→    }
  4258→  }
  4259→  return row;
  4260→}
