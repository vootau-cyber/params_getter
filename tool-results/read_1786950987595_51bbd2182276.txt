    63→        if (isNaN(n) || n < -90 || n > 90) return 'invalid';
    64→        continue;
    65→      }

    67→        var n2 = parseFloat(str);
    68→        if (isNaN(n2) || n2 < -180 || n2 > 180) return 'invalid';
    69→        continue;
    70→      }
    71→      if (rule.minLen !== undefined) {
    72→        if (str.length < rule.minLen) return 'invalid';
    73→        continue;
    74→      }
    75→      if (rule.test && !rule.test.test(str)) return 'invalid';
    76→    }
    77→    return 'ok';

   177→  function esc(str) {
   178→    if (str === null || str === undefined) return '';
   179→    return String(str)
   180→      .replace(/&/g, '&amp;')
   181→      .replace(/</g, '&lt;')
   182→      .replace(/>/g, '&gt;')
   183→      .replace(/"/g, '&quot;');
   184→  }

   186→  function deepClone(obj) {
   187→    return JSON.parse(JSON.stringify(obj));
   188→  }

   190→  function getSectionDef(sectionKey) {
   191→    if (!APP.schema || !APP.schema.sections) return null;
   192→    for (var i = 0; i < APP.schema.sections.length; i++) {
   193→      if (APP.schema.sections[i].key === sectionKey) return APP.schema.sections[i];
   194→    }
   195→    return null;
   196→  }
   197→
   198→  function getFieldDef(sectionKey, fieldKey) {
   199→    var sec = getSectionDef(sectionKey);
   200→    if (!sec || !sec.fields) return null;
   201→    for (var i = 0; i < sec.fields.length; i++) {
   202→      if (sec.fields[i].key === fieldKey) return sec.fields[i];
   203→    }
   204→    return null;
   205→  }
   206→
   207→  function isVirtualField(sectionKey, fieldKey) {
   208→    var fd = getFieldDef(sectionKey, fieldKey);
   209→    return fd && fd.virtual === true;
   210→  }

   212→  function stripVirtuals(data) {
   213→    var cleaned = deepClone(data);
   214→    if (!APP.schema || !APP.schema.sections) return cleaned;
   215→    for (var s = 0; s < APP.schema.sections.length; s++) {
   216→      var secKey = APP.schema.sections[s].key;
   217→      if (!cleaned[secKey]) continue;
   218→      var virtualKeys = [];
   219→      for (var f = 0; f < APP.schema.sections[s].fields.length; f++) {
   220→        if (APP.schema.sections[s].fields[f].virtual) {
   221→          virtualKeys.push(APP.schema.sections[s].fields[f].key);
   222→        }
   223→      }
   224→      for (var r = 0; r < cleaned[secKey].length; r++) {
   225→        for (var vk = 0; vk < virtualKeys.length; vk++) {
   226→          delete cleaned[secKey][r][virtualKeys[vk]];
   227→        }
   228→      }
   229→    }
   230→    return cleaned;
   231→  }

   421→      var srcKey = mapping[targetKey];

   497→  function renderField(sectionKey, rowIndex, fieldDef, value) {
   498→    var key = fieldDef.key;
   499→    var label = fieldDef.label || key;
   500→    var type = fieldDef.type || 'text';
   501→    var readOnly = fieldDef.readOnly === true;
   502→    var autoFilled = fieldDef.autoFilled === true;
   503→    var isVirtual = fieldDef.virtual === true;
   504→    var options = fieldDef.options || [];
   505→    var nestedFields = fieldDef.nestedFields || [];
   506→    var refSection = fieldDef.refSection;
   507→    var refLabelField = fieldDef.refLabelField;
   508→
   509→    var status = validateField(key, value);
   510→    var statusClass = '';
   511→    var tooltipHtml = '';
   512→
   513→    if (status === 'empty') {
   514→      statusClass = ' field-status-empty';
   515→    } else if (status === 'invalid') {
   516→      statusClass = ' invalid-field';
   517→      var hint = getFieldHint(key);
   518→      tooltipHtml = '<span class="invalid-tooltip">Поле ' + esc(label) + ' — некорректный формат ввода. Корректный формат: ' + esc(hint) + '</span>';
   519→    }
   520→
   521→    var autoBadge = '';
   522→    if (autoFilled || isVirtual) {
   523→      autoBadge = ' <span class="badge-auto">авто</span>';
   524→    }
   525→
   526→    var dataAttrs = ' data-section="' + esc(sectionKey) + '" data-row="' + rowIndex + '" data-field="' + esc(key) + '"';
   527→
   528→    var labelHtml = '<label class="field-label" style="display:flex;align-items:center;gap:6px;">' +
   529→      (status === 'empty' && !readOnly ? '<span style="width:6px;height:6px;border-radius:50%;background:var(--destructive);flex-shrink:0;"></span>' : '') +
   530→      esc(label) +
   531→      autoBadge +
   532→      tooltipHtml +
   533→    '</label>';
   534→
   535→    var inputHtml = '';
   536→
   537→    if (type === 'object') {
   538→      var nestedData = Array.isArray(value) ? value : [];
   539→      inputHtml = renderNestedTable(sectionKey, rowIndex, key, label, nestedFields, nestedData, readOnly);
   540→      return '<div class="field-wrapper" style="margin-bottom:16px;">' + inputHtml + '</div>';
   541→    }
   542→
   543→    var disabledAttr = readOnly ? ' disabled' : '';
   544→    var roClass = readOnly ? ' style="opacity:0.6;background:var(--muted);"' : '';
   545→
   546→    if (type === 'boolean') {
   547→      var checked = value === true || value === 'true' || value === 1 ? ' checked' : '';
   548→      inputHtml = '<div class="field-checkbox-wrapper">' +
   549→        '<input type="checkbox" class="field-input"' + dataAttrs + checked + disabledAttr + ' />' +
   550→      '</div>';
   551→    } else if (type === 'select' && options.length > 0) {
   552→      inputHtml = '<select class="field-select' + statusClass + '"' + dataAttrs + disabledAttr + roClass + '>';
   553→      inputHtml += '<option value=""></option>';
   554→      for (var o = 0; o < options.length; o++) {
   555→        var opt = options[o];
   556→        var optVal = typeof opt === 'object' ? opt.value : opt;
   557→        var optLabel = typeof opt === 'object' ? opt.label : opt;
   558→        var sel = (String(value) === String(optVal)) ? ' selected' : '';
   559→        inputHtml += '<option value="' + esc(optVal) + '"' + sel + '>' + esc(optLabel) + '</option>';
   560→      }
   561→      inputHtml += '</select>';
   562→    } else if (type === 'ref' || type === 'virtual') {
   563→      inputHtml = renderRefSelect(sectionKey, rowIndex, key, fieldDef, value, statusClass, disabledAttr, roClass, isVirtual);
   564→    } else if (type === 'textarea') {
   565→      inputHtml = '<textarea class="field-textarea' + statusClass + '"' + dataAttrs + disabledAttr + roClass + ' rows="3">' + esc(value) + '</textarea>';
   566→    } else if (type === 'number') {
   567→      inputHtml = '<input type="number" class="field-number' + statusClass + '"' + dataAttrs + ' value="' + esc(value) + '"' + disabledAttr + roClass + ' />';
   568→    } else if (type === 'date') {
   569→      var dateVal = value ? String(value).substring(0, 10) : '';
   570→      inputHtml = '<input type="date" class="field-date' + statusClass + '"' + dataAttrs + ' value="' + esc(dateVal) + '"' + disabledAttr + roClass + ' />';
   571→    } else if (type === 'array') {
   572→      var arrStr = Array.isArray(value) ? value.join(', ') : (value || '');
   573→      inputHtml = '<input type="text" class="field-input' + statusClass + '"' + dataAttrs + ' value="' + esc(arrStr) + '"' + disabledAttr + roClass + ' placeholder="Значения через запятую" />';
   574→    } else {
   575→      inputHtml = '<input type="text" class="field-input' + statusClass + '"' + dataAttrs + ' value="' + esc(value) + '"' + disabledAttr + roClass + ' />';
   576→    }
   577→
   578→    return '<div class="field-wrapper" style="margin-bottom:16px;">' + labelHtml + inputHtml + '</div>';
   579→  }

   680→  function pluralRu(n, one, few, many) {
   681→    var abs = Math.abs(n) % 100;
   682→    var lastDigit = abs % 10;
   683→    if (abs > 10 && abs < 20) return many;
   684→    if (lastDigit > 1 && lastDigit < 5) return few;
   685→    if (lastDigit === 1) return one;
   686→    return many;
   687→  }

  1134→  function handleFieldInput(sectionKey, rowIndex, fieldKey, target, e) {
  1135→    if (!APP.data[sectionKey] || !APP.data[sectionKey][rowIndex]) return;
  1136→    var fd = getFieldDef(sectionKey, fieldKey);
  1137→    if (!fd) return;
  1138→    var type = fd.type || 'text';
  1139→
  1140→    if (type === 'boolean') {
  1141→      APP.data[sectionKey][rowIndex][fieldKey] = target.checked;
  1142→    } else if (type === 'number') {
  1143→      var numVal = target.value === '' ? null : parseFloat(target.value);
  1144→      APP.data[sectionKey][rowIndex][fieldKey] = numVal;
  1145→    } else if (type === 'array') {
  1146→      var strVal = target.value.trim();
  1147→      APP.data[sectionKey][rowIndex][fieldKey] = strVal ? strVal.split(',').map(function (s) { return s.trim(); }).filter(function (s) { return s !== ''; }) : [];
  1148→    } else {
  1149→      APP.data[sectionKey][rowIndex][fieldKey] = target.value;
  1150→    }
  1151→
  1152→    checkDirty();
  1153→  }
  1154→
  1155→  function handleFieldChange(sectionKey, rowIndex, fieldKey, target, e) {
  1156→    if (!APP.data[sectionKey] || !APP.data[sectionKey][rowIndex]) return;
  1157→    var fd = getFieldDef(sectionKey, fieldKey);
  1158→    if (!fd) return;
  1159→    var type = fd.type || 'text';
  1160→
  1161→    if (type === 'boolean') {
  1162→      APP.data[sectionKey][rowIndex][fieldKey] = target.checked;
  1163→      checkDirty();
  1164→      return;
  1165→    }
  1166→
  1167→    if (type === 'ref') {
  1168→      APP.data[sectionKey][rowIndex][fieldKey] = target.value;
  1169→      /* Check infra ref auto-fill */
  1170→      if (INFRA_REF_MAP[fieldKey]) {
  1171→        handleInfraRefChange(sectionKey, rowIndex, fieldKey, target.value);
  1172→      }
  1173→      checkDirty();
  1174→      return;
  1175→    }
  1176→
  1177→    if (type === 'virtual') {
  1178→      APP.data[sectionKey][rowIndex][fieldKey] = target.value;
  1179→      handleVirtualRefChange(sectionKey, rowIndex, fieldKey, target.value);
  1180→      checkDirty();
  1181→      renderApp();
  1182→      return;
  1183→    }
  1184→
  1185→    if (type === 'select') {
  1186→      APP.data[sectionKey][rowIndex][fieldKey] = target.value;
  1187→      checkDirty();
  1188→      return;
  1189→    }
  1190→
  1191→    /* For number type on change (spin buttons, etc.) */
  1192→    if (type === 'number') {
  1193→      var numVal2 = target.value === '' ? null : parseFloat(target.value);
  1194→      APP.data[sectionKey][rowIndex][fieldKey] = numVal2;
  1195→      checkDirty();
  1196→      return;
  1197→    }
  1198→  }

  1200→  function handleNestedInput(sectionKey, rowIndex, fieldKey, nestedRowIndex, nestedFieldKey, target, e) {
  1201→    if (!APP.data[sectionKey] || !APP.data[sectionKey][rowIndex]) return;
  1202→    var arr = APP.data[sectionKey][rowIndex][fieldKey];
  1203→    if (!Array.isArray(arr) || !arr[nestedRowIndex]) return;
  1204→
  1205→    var fd = getFieldDef(sectionKey, fieldKey);
  1206→    var nfDef = null;
  1207→    if (fd && fd.nestedFields) {
  1208→      for (var i = 0; i < fd.nestedFields.length; i++) {
  1209→        if (fd.nestedFields[i].key === nestedFieldKey) { nfDef = fd.nestedFields[i]; break; }
  1210→      }
  1211→    }
  1212→
  1213→    var nType = nfDef ? nfDef.type : 'text';
  1214→    if (nType === 'boolean') {
  1215→      arr[nestedRowIndex][nestedFieldKey] = target.checked;
  1216→    } else if (nType === 'number') {
  1217→      arr[nestedRowIndex][nestedFieldKey] = target.value === '' ? null : parseFloat(target.value);
  1218→    } else {
  1219→      arr[nestedRowIndex][nestedFieldKey] = target.value;
  1220→    }
  1221→    checkDirty();
  1222→  }
