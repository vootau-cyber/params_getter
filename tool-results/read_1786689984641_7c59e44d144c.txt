    63→/** Simple debounce utility */
    64→function debounce<T extends (...args: Parameters<T>) => void>(
    65→  fn: T,
    66→  ms: number,
    67→): (...args: Parameters<T>) => void {
    68→  let timer: ReturnType<typeof setTimeout>;
    69→  return (...args: Parameters<T>) => {
    70→    clearTimeout(timer);
    71→    timer = setTimeout(() => fn(...args), ms);
    72→  };
    73→}

   216→export type FieldStatus = 'empty' | 'invalid' | 'ok';

   218→/**
   219→ * Find the matching format rule for a field key, or null if none.
   220→ */
   221→function findFormatRule(fieldKey: string): FormatRule | null {
   222→  for (const rule of FORMAT_RULES) {
   223→    if (rule.keyPattern.test(fieldKey)) return rule;
   224→  }
   225→  return null;
   226→}
   227→
   228→/**
   229→ * Get the Russian format hint for a field (for tooltips).
   230→ * Returns null if the field has no specific format requirement.
   231→ */
   232→export function getFieldFormatHint(field: FieldDef): string | null {
   233→  // Number fields: always validate as finite number
   234→  if (field.type === 'number') {
   235→    return 'Число (дробное через точку)';
   236→  }
   237→  // Date fields: always validate as YYYY-MM-DD
   238→  if (field.type === 'date') {
   239→    return 'ГГГГ-ММ-ДД (например, 2024-12-31)';
   240→  }
   241→  // Text/textarea: check against FORMAT_RULES by key name
   242→  const rule = findFormatRule(field.key);
   243→  return rule ? rule.formatHint : null;
   244→}
   245→
   246→/**
   247→ * Determine if a field value is "empty" or "invalid".
   248→ * Booleans, readOnly, virtual, and object fields are always 'ok'.
   249→ *
   250→ * Returns:
   251→ *   'empty'   — value is missing / unfilled
   252→ *   'invalid' — value is present but fails format validation
   253→ *   'ok'      — value is present and valid
   254→ */
   255→export function getFieldStatus(field: FieldDef, value: unknown): FieldStatus {
   256→  // Skip types that don't need validation
   257→  if (field.readOnly || field.virtual) return 'ok';
   258→  if (field.type === 'boolean') return 'ok';
   259→  if (field.type === 'object') return 'ok';
   260→
   261→  // Helper: is the value considered "empty" for this type?
   262→  const isEmpty = (): boolean => {
   263→    switch (field.type) {
   264→      case 'text':
   265→      case 'textarea':
   266→        return (value as string) === '' || value === undefined || value === null;
   267→      case 'number':
   268→        return value === null || value === undefined;
   269→      case 'date':
   270→        return value === null || value === undefined || value === '';
   271→      case 'select':
   272→        return (value as string) === '' || value === null;
   273→      case 'array':
   274→        return !Array.isArray(value) || value.length === 0;
   275→      case 'ref':
   276→        return value === null || value === undefined;
   277→      default:
   278→        return false;
   279→    }
   280→  };
   281→
   282→  if (isEmpty()) return 'empty';
   283→
   284→  // ── Format validation for non-empty values ──
   285→
   286→  // Number: must be a finite number
   287→  if (field.type === 'number') {
   288→    if (typeof value !== 'number' || !Number.isFinite(value)) return 'invalid';
   289→    return 'ok';
   290→  }
   291→
   292→  // Date: must match YYYY-MM-DD and be a real date
   293→  if (field.type === 'date') {
   294→    const str = String(value);
   295→    if (!/^\d{4}-\d{2}-\d{2}$/.test(str)) return 'invalid';
   296→    const d = new Date(str + 'T00:00:00');
   297→    if (isNaN(d.getTime())) return 'invalid';
   298→    // Verify the parsed date matches the input (e.g. no rollover like 2024-02-30)
   299→    const [y, m, day] = str.split('-').map(Number);
   300→    if (d.getFullYear() !== y || d.getMonth() + 1 !== m || d.getDate() !== day) {
   301→      return 'invalid';
   302→    }
   303→    return 'ok';
   304→  }
   305→
   306→  // Text / textarea: check against format rules by key name
   307→  if (field.type === 'text' || field.type === 'textarea') {
   308→    const str = String(value);
   309→    const rule = findFormatRule(field.key);
   310→    if (rule && !rule.validate(str)) return 'invalid';
   311→    return 'ok';
   312→  }
   313→
   314→  // Select: if value is set, it's always valid (chosen from a list)
   315→  if (field.type === 'select') return 'ok';
   316→
   317→  // Array: if non-empty, valid
   318→  if (field.type === 'array') return 'ok';
   319→
   320→  // Ref: if set, valid
   321→  if (field.type === 'ref') return 'ok';
   322→
   323→  return 'ok';
   324→}

   326→/** Count empty + invalid fields in a record */
   327→export function countFieldIssues(
   328→  fields: FieldDef[],
   329→  row: Record<string, unknown>,
   330→): number {
   331→  let count = 0;
   332→  for (const f of fields) {
   333→    const status = getFieldStatus(f, row[f.key]);
   334→    if (status !== 'ok') count++;
   335→  }
   336→  return count;
   337→}

   606→
   607→  const isAutoFilled = field.autoFilled;
   608→  const status = getFieldStatus(field, value);
   609→  const hasIssue = status !== 'ok';
   610→
   611→  const labelContent = (
   612→    <div className="flex items-center gap-1.5">
   613→      {hasIssue && (
   614→        <AlertCircle className="size-3.5 text-destructive shrink-0" />
   615→      )}
   616→      <Label
   617→        className={`text-xs font-medium ${isAutoFilled ? 'text-primary' : hasIssue ? 'text-destructive' : 'text-foreground/80'}`}
   618→      >
   619→        {field.label}
   620→      </Label>
   621→      {isAutoFilled && (
   622→        <Badge
   623→          variant="outline"
   624→          className="text-[9px] px-1 py-0 h-3.5 text-primary/70 border-primary/30"
   625→        >
   626→          авто
   627→        </Badge>
   628→      )}
   629→      {field.hint && (
   630→        <Tooltip>
   631→          <TooltipTrigger asChild>
   632→            <span className="text-muted-foreground/50 cursor-help text-[10px] leading-none">
   633→              ?
   634→            </span>
   635→          </TooltipTrigger>
   636→          <TooltipContent side="top" sideOffset={4} className="max-w-xs">
   637→            <p className="text-xs">{field.hint}</p>
   638→          </TooltipContent>
   639→        </Tooltip>
   640→      )}
   641→    </div>
   642→  );
   643→
   644→  const inputEl = (
   645→    <FieldInput
   646→      field={field}
   647→      value={value}
   648→      sectionKey={sectionKey}
   649→      rowIndex={rowIndex}
   650→      status={status}
   651→      updateCell={updateCell}
   652→      updateNestedCell={updateNestedCell}
   653→      addNestedRow={addNestedRow}
   654→      removeNestedRow={removeNestedRow}
   655→    />
   656→  );
   657→
   658→  // For object fields (nested tables), render differently
   659→  if (field.type === 'object' && field.nestedFields) {
   660→    return (
   661→      <div className={fullWidth ? '' : 'col-span-2'}>
   662→        <div className="flex items-center gap-2 mb-2">{labelContent}</div>
   663→        <NestedFieldBlock
   664→          field={field}
   665→          value={value}
   666→          sectionKey={sectionKey}
   667→          rowIndex={rowIndex}
   668→          updateNestedCell={updateNestedCell}
   669→          addNestedRow={addNestedRow}
   670→          removeNestedRow={removeNestedRow}
   671→        />
   672→      </div>
   673→    );
   674→  }
   675→
   676→  // For array fields that need more space
   677→  if (field.type === 'array' && fullWidth) {
   678→    return (
   679→      <div>
   680→        {labelContent}
   681→        <div className="mt-1">{inputEl}</div>
   682→      </div>
   683→    );
   684→  }
   685→
   686→  // For textarea in grid, span 2 columns
   687→  if (field.type === 'textarea') {
   688→    return (
   689→      <div className="col-span-2">
   690→        {labelContent}
   691→        <div className="mt-1">{inputEl}</div>
   692→      </div>
   693→    );
   694→  }
   695→
   696→  return (
   697→    <div>
   698→      {labelContent}
   699→      <div className="mt-1">{inputEl}</div>
   700→    </div>
   701→  );
   702→}

   773→
   774→function FieldInput({
   775→  field,
   776→  value,
   777→  sectionKey,
   778→  rowIndex,
   779→  status,
   780→  updateCell,
   781→}: FieldInputProps) {
   782→  const isAutoFilled = field.autoFilled;
   783→  const hasIssue = status !== 'ok';
   784→  const isInvalid = status === 'invalid';
   785→
   786→  // Build class based on status
   787→  const baseInputClass = isAutoFilled
   788→    ? 'h-9 text-sm border-primary/25 bg-primary/5 focus-visible:ring-primary/40'
   789→    : hasIssue
   790→      ? 'h-9 text-sm border-destructive/50 bg-destructive/5 focus-visible:ring-destructive/40'
   791→      : 'h-9 text-sm focus-visible:ring-1 focus-visible:ring-ring';
   792→
   793→  switch (field.type) {
   794→    case 'text': {
   795→      const inputEl = (
   796→        <Input
   797→          type="text"
   798→          value={(value as string) || ''}
   799→          placeholder={field.placeholder}
   800→          className={baseInputClass}
   801→          onChange={(e) =>
   802→            updateCell(sectionKey, rowIndex, field.key, e.target.value)
   803→          }
   804→        />
   805→      );
   806→      return isInvalid ? (
   807→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   808→      ) : (
   809→        inputEl
   810→      );
   811→    }
   812→
   813→    case 'textarea': {
   814→      const taClass = hasIssue
   815→        ? 'min-h-[80px] text-sm resize-y border-destructive/50 bg-destructive/5 focus-visible:ring-destructive/40'
   816→        : 'min-h-[80px] text-sm resize-y focus-visible:ring-1 focus-visible:ring-ring';
   817→      const inputEl = (
   818→        <Textarea
   819→          value={(value as string) || ''}
   820→          placeholder={field.placeholder}
   821→          rows={3}
   822→          className={taClass}
   823→          onChange={(e) =>
   824→            updateCell(sectionKey, rowIndex, field.key, e.target.value)
   825→          }
   826→        />
   827→      );
   828→      return isInvalid ? (
   829→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   830→      ) : (
   831→        inputEl
   832→      );
   833→    }
   834→
   835→    case 'number': {
   836→      const inputEl = (
   837→        <Input
   838→          type="number"
   839→          step="any"
   840→          value={
   841→            value === null || value === undefined ? '' : String(value)
   842→          }
   843→          placeholder={field.placeholder}
   844→          className={baseInputClass}
   845→          onChange={(e) => {
   846→            const v = e.target.value;
   847→            updateCell(
   848→              sectionKey,
   849→              rowIndex,
   850→              field.key,
   851→              v === '' ? null : Number(v),
   852→            );
   853→          }}
   854→        />
   855→      );
   856→      return isInvalid ? (
   857→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   858→      ) : (
   859→        inputEl
   860→      );
   861→    }
   862→
   863→    case 'date': {
   864→      const inputEl = (
   865→        <Input
   866→          type="date"
   867→          value={(value as string) || ''}
   868→          className={baseInputClass}
   869→          onChange={(e) =>
   870→            updateCell(
   871→              sectionKey,
   872→              rowIndex,
   873→              field.key,
   874→              e.target.value || null,
   875→            )
   876→          }
   877→        />
   878→      );
   879→      return isInvalid ? (
   880→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   881→      ) : (
   882→        inputEl
   883→      );
   884→    }
   885→
   886→    case 'boolean':
   887→      return (
   888→        <div className="flex items-center h-9">
   889→          <Switch
   890→            checked={!!value}
   891→            onCheckedChange={(checked) =>
   892→              updateCell(sectionKey, rowIndex, field.key, checked)
   893→            }
   894→          />
   895→        </div>
   896→      );
   897→
   898→    case 'select': {
   899→      const selectEl = (
   900→        <Select
   901→          value={(value as string) || ''}
   902→          onValueChange={(v) =>
   903→            updateCell(sectionKey, rowIndex, field.key, v)
   904→          }
   905→        >
   906→          <SelectTrigger
   907→            className={
   908→              isAutoFilled
   909→                ? 'h-9 text-sm border-primary/25 bg-primary/5 focus:ring-primary/40 w-full'
   910→                : hasIssue
   911→                  ? 'h-9 text-sm border-destructive/50 bg-destructive/5 focus:ring-destructive/40 w-full'
   912→                  : 'h-9 text-sm w-full'
   913→            }
   914→          >
   915→            <SelectValue placeholder={field.placeholder || 'Выберите…'} />
   916→          </SelectTrigger>
   917→          <SelectContent>
   918→            {(field.options || []).map((opt) => (
   919→              <SelectItem key={opt} value={opt}>
   920→                {opt}
   921→              </SelectItem>
   922→            ))}
   923→          </SelectContent>
   924→        </Select>
   925→      );
   926→      return isInvalid ? (
   927→        <InvalidFieldTooltip field={field}>{selectEl}</InvalidFieldTooltip>
   928→      ) : (
   929→        selectEl
   930→      );
   931→    }
   932→
   933→    case 'array': {
   934→      const arr = Array.isArray(value) ? (value as string[]) : [];
   935→      const inputEl = (
   936→        <Input
   937→          type="text"
   938→          value={arr.join(', ')}
   939→          placeholder="через запятую"
   940→          className={baseInputClass}
   941→          onChange={(e) => {
   942→            const items = e.target.value
   943→              .split(',')
   944→              .map((s) => s.trim())
   945→              .filter(Boolean);
   946→            updateCell(sectionKey, rowIndex, field.key, items);
   947→          }}
   948→        />
   949→      );
   950→      return isInvalid ? (
   951→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   952→      ) : (
   953→        inputEl
   954→      );
   955→    }
   956→
   957→    case 'ref': {
   958→      const refEl = (
   959→        <RefSelect
   960→          field={field}
   961→          value={value}
   962→          sectionKey={sectionKey}
   963→          rowIndex={rowIndex}
   964→          hasIssue={hasIssue}
   965→        />
   966→      );
   967→      return isInvalid ? (
   968→        <InvalidFieldTooltip field={field}>{refEl}</InvalidFieldTooltip>
   969→      ) : (
   970→        refEl
   971→      );
   972→    }
   973→
   974→    default:
   975→      return <span className="text-muted-foreground text-xs">—</span>;
   976→  }
   977→}

   979→// =============================================================================
   980→// RefSelect — dropdown populated from another section's data
   981→// =============================================================================
   982→
   983→function RefSelect({
   984→  field,
   985→  value,
   986→  sectionKey,
   987→  rowIndex,
   988→  hasIssue,
   989→}: {
   990→  field: FieldDef;
   991→  value: unknown;
   992→  sectionKey: string;
   993→  rowIndex: number;
   994→  hasIssue: boolean;
   995→}) {
   996→  const updateCell = useStore((s) => s.updateCell);
   997→  const allData = useStore((s) => s.data);
   998→  const refSection = field.refSection;
   999→  const refLabelField = field.refLabelField;
  1000→
  1001→  const options = useMemo(() => {
  1002→    if (!refSection) return [];
  1003→    const rows = (allData[refSection] || []) as Record<string, unknown>[];
  1004→    return rows
  1005→      .map((row, idx) => {
  1006→        const label = refLabelField
  1007→          ? String(row[refLabelField] || '')
  1008→          : `Запись ${idx + 1}`;
  1009→        return { idx, label };
  1010→      })
  1011→      .filter((o) => o.label);
  1012→  }, [allData, refSection, refLabelField]);
  1013→
  1014→  const currentIdx = value as number | null;
  1015→  const currentValue =
  1016→    currentIdx !== null && currentIdx !== undefined
  1017→      ? String(currentIdx)
  1018→      : '';
  1019→  const currentLabel =
  1020→    currentIdx !== null && currentIdx !== undefined
  1021→      ? options.find((o) => o.idx === currentIdx)?.label || ''
  1022→      : '';
  1023→
  1024→  const isVirtual = field.virtual;
  1025→  const triggerClass = hasIssue
  1026→    ? 'h-9 text-sm w-full border-destructive/50 bg-destructive/5 focus:ring-destructive/40'
  1027→    : isVirtual
  1028→      ? 'h-9 text-sm w-full border-primary/30 bg-primary/5 focus:ring-1 focus:ring-primary'
  1029→      : 'h-9 text-sm w-full';
  1030→
  1031→  return (
  1032→    <Select
  1033→      value={currentValue}
  1034→      onValueChange={(v) =>
  1035→        updateCell(sectionKey, rowIndex, field.key, Number(v))
  1036→      }
  1037→    >
  1038→      <SelectTrigger className={triggerClass}>
  1039→        <SelectValue placeholder={field.placeholder || 'Выберите…'}>
  1040→          {currentLabel || (field.placeholder || 'Выберите…')}
  1041→        </SelectValue>
  1042→      </SelectTrigger>
  1043→      <SelectContent>
  1044→        {options.length === 0 && (
  1045→          <SelectItem value="_empty" disabled>
  1046→            Нет записей в справочнике
  1047→          </SelectItem>
  1048→        )}
  1049→        {options.map((opt) => (
  1050→          <SelectItem key={opt.idx} value={String(opt.idx)}>
  1051→            {opt.label}
  1052→          </SelectItem>
  1053→        ))}
  1054→      </SelectContent>
  1055→    </Select>
  1056→  );
  1057→}

  1060→// NestedFieldBlock — for object fields with nested sub-fields

  1062→
  1063→function NestedFieldBlock({
  1064→  field,
  1065→  value,
  1066→  sectionKey,
  1067→  rowIndex,
  1068→  updateNestedCell,
  1069→  addNestedRow,
  1070→  removeNestedRow,
  1071→}: {
  1072→  field: FieldDef;
  1073→  value: unknown;
  1074→  sectionKey: string;
  1075→  rowIndex: number;
  1076→  updateNestedCell: (
  1077→    sectionKey: string,
  1078→    rowIndex: number,
  1079→    fieldKey: string,
  1080→    nestedIndex: number,
  1081→    nestedFieldKey: string,
  1082→    value: unknown,
  1083→  ) => void;
  1084→  addNestedRow: (
  1085→    sectionKey: string,
  1086→    rowIndex: number,
  1087→    fieldKey: string,
  1088→  ) => void;
  1089→  removeNestedRow: (
  1090→    sectionKey: string,
  1091→    rowIndex: number,
  1092→    fieldKey: string,
  1093→    nestedIndex: number,
  1094→  ) => void;
  1095→}) {
  1096→  const [open, setOpen] = useState(false);
  1097→  const editableNestedFields = (field.nestedFields || []).filter(
  1098→    (nf) => !nf.readOnly,
  1099→  );
  1100→
  1101→  // Array of nested objects (e.g., points, scenarios, cargo lists)
  1102→  if (Array.isArray(value)) {
  1103→    const items = value as Record<string, unknown>[];
  1104→    return (
  1105→      <Collapsible open={open} onOpenChange={setOpen}>
  1106→        <CollapsibleTrigger asChild>
  1107→          <button className="flex items-center gap-2 text-xs text-muted-foreground hover:text-foreground transition-colors">
  1108→            {open ? (
  1109→              <ChevronDown className="size-3.5" />
  1110→            ) : (
  1111→              <ChevronRight className="size-3.5" />
  1112→            )}
  1113→            <Badge variant="secondary" className="text-xs">
  1114→              {items.length}{' '}
  1115→              {items.length === 1
  1116→                ? 'запись'
  1117→                : items.length < 5
  1118→                  ? 'записи'
  1119→                  : 'записей'}
  1120→            </Badge>
  1121→            <span className="underline">развернуть</span>
  1122→          </button>
  1123→        </CollapsibleTrigger>
  1124→        <CollapsibleContent>
  1125→          <div className="mt-2 border rounded-lg overflow-hidden">
  1126→            {/* Header */}
  1127→            <div className="grid bg-muted/50 border-b px-3 py-2 text-xs font-medium text-muted-foreground">
  1128→              <div className="flex gap-4">
  1129→                <span className="w-8 text-center">#</span>
  1130→                {editableNestedFields.map((nf) => (
  1131→                  <span key={nf.key} className="min-w-[120px]">
  1132→                    {nf.label}
  1133→                  </span>
  1134→                ))}
  1135→                <span className="w-7" />
  1136→              </div>
  1137→            </div>
  1138→            {/* Rows */}
  1139→            {items.map((item, ni) => (
  1140→              <div
  1141→                key={ni}
  1142→                className="grid border-b last:border-b-0 hover:bg-muted/20 px-3 py-1.5"
  1143→              >
  1144→                <div className="flex gap-4 items-center">
  1145→                  <span className="w-8 text-center text-xs text-muted-foreground">
  1146→                    {ni + 1}
  1147→                  </span>
  1148→                  {editableNestedFields.map((nf) => (
  1149→                    <div key={nf.key} className="min-w-[120px]">
  1150→                      <NestedFieldInput
  1151→                        nestedField={nf}
  1152→                        value={item[nf.key]}
  1153→                        onChange={(v) =>
  1154→                          updateNestedCell(
  1155→                            sectionKey,
  1156→                            rowIndex,
  1157→                            field.key,
  1158→                            ni,
  1159→                            nf.key,
  1160→                            v,
  1161→                          )
  1162→                        }
  1163→                      />
  1164→                    </div>
  1165→                  ))}
  1166→                  <div className="w-7">
  1167→                    <Button
  1168→                      variant="ghost"
  1169→                      size="icon"
  1170→                      className="size-6 text-muted-foreground hover:text-destructive"
  1171→                      onClick={() =>
  1172→                        removeNestedRow(
  1173→                          sectionKey,
  1174→                          rowIndex,
  1175→                          field.key,
  1176→                          ni,
  1177→                        )
  1178→                      }
  1179→                    >
  1180→                      <Minus className="size-3" />
  1181→                    </Button>
  1182→                  </div>
  1183→                </div>
  1184→              </div>
  1185→            ))}
  1186→            {/* Add row */}
  1187→            <div className="px-3 py-1.5 border-t">
  1188→              <Button
  1189→                variant="ghost"
  1190→                size="sm"
  1191→                className="h-7 text-xs"
  1192→                onClick={() =>
  1193→                  addNestedRow(sectionKey, rowIndex, field.key)
  1194→                }
  1195→              >
  1196→                <Plus className="mr-1 size-3" />
  1197→                Добавить
  1198→              </Button>
  1199→            </div>
  1200→          </div>
  1201→        </CollapsibleContent>
  1202→      </Collapsible>
  1203→    );
  1204→  }
  1205→
  1206→  // Single nested object (e.g., ztb_boundaries)
  1207→  if (field.nestedFields && typeof value === 'object' && value !== null) {
  1208→    const obj = value as Record<string, unknown>;
  1209→    return (
  1210→      <div className="grid grid-cols-2 gap-x-6 gap-y-3">
  1211→        {editableNestedFields.map((nf) => (
  1212→          <div key={nf.key}>
  1213→            <Label className="text-xs font-medium text-foreground/80">
  1214→              {nf.label}
  1215→            </Label>
  1216→            <div className="mt-1">
  1217→              <NestedFieldInput
  1218→                nestedField={nf}
  1219→                value={obj[nf.key]}
  1220→                onChange={(v) => {
  1221→                  const updated = { ...obj, [nf.key]: v };
  1222→                  useStore
  1223→                    .getState()
  1224→                    .updateCell(sectionKey, rowIndex, field.key, updated);
  1225→                }}
  1226→              />
  1227→            </div>
  1228→          </div>
  1229→        ))}
  1230→      </div>
  1231→    );
  1232→  }
  1233→
  1234→  return null;
  1235→}

  1237→// =============================================================================
  1238→// NestedFieldInput — input for a single nested field
  1239→// =============================================================================
  1240→
  1241→function NestedFieldInput({
  1242→  nestedField,
  1243→  value,
  1244→  onChange,
  1245→}: {
  1246→  nestedField: FieldDef;
  1247→  value: unknown;
  1248→  onChange: (value: unknown) => void;
  1249→}) {
  1250→  const v = value ?? '';
  1251→
  1252→  switch (nestedField.type) {
  1253→    case 'boolean':

  1295→          onChange={(e) => onChange(e.target.value)}
  1296→        />
  1297→      );
  1298→  }
