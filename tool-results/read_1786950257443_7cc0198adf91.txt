    64→/** Simple debounce utility */
    65→function debounce<T extends (...args: Parameters<T>) => void>(
    66→  fn: T,
    67→  ms: number,
    68→): (...args: Parameters<T>) => void {
    69→  let timer: ReturnType<typeof setTimeout>;
    70→  return (...args: Parameters<T>) => {
    71→    clearTimeout(timer);
    72→    timer = setTimeout(() => fn(...args), ms);
    73→  };
    74→}

   217→export type FieldStatus = 'empty' | 'invalid' | 'ok';
   218→
   219→/**
   220→ * Find the matching format rule for a field key, or null if none.
   221→ */
   222→function findFormatRule(fieldKey: string): FormatRule | null {
   223→  for (const rule of FORMAT_RULES) {
   224→    if (rule.keyPattern.test(fieldKey)) return rule;
   225→  }
   226→  return null;
   227→}
   228→
   229→/**
   230→ * Get the Russian format hint for a field (for tooltips).
   231→ * Returns null if the field has no specific format requirement.
   232→ */
   233→export function getFieldFormatHint(field: FieldDef): string | null {
   234→  // Number fields: always validate as finite number
   235→  if (field.type === 'number') {
   236→    return 'Число (дробное через точку)';
   237→  }
   238→  // Date fields: always validate as YYYY-MM-DD
   239→  if (field.type === 'date') {
   240→    return 'ГГГГ-ММ-ДД (например, 2024-12-31)';
   241→  }
   242→  // Text/textarea: check against FORMAT_RULES by key name
   243→  const rule = findFormatRule(field.key);
   244→  return rule ? rule.formatHint : null;
   245→}
   246→
   247→/**
   248→ * Determine if a field value is "empty" or "invalid".
   249→ * Booleans, readOnly, virtual, and object fields are always 'ok'.
   250→ *
   251→ * Returns:
   252→ *   'empty'   — value is missing / unfilled
   253→ *   'invalid' — value is present but fails format validation
   254→ *   'ok'      — value is present and valid
   255→ */
   256→export function getFieldStatus(field: FieldDef, value: unknown): FieldStatus {
   257→  // Skip types that don't need validation
   258→  if (field.readOnly || field.virtual) return 'ok';
   259→  if (field.type === 'boolean') return 'ok';
   260→  if (field.type === 'object') return 'ok';
   261→
   262→  // Helper: is the value considered "empty" for this type?
   263→  const isEmpty = (): boolean => {
   264→    switch (field.type) {
   265→      case 'text':
   266→      case 'textarea':
   267→        return (value as string) === '' || value === undefined || value === null;
   268→      case 'number':
   269→        return value === null || value === undefined;
   270→      case 'date':
   271→        return value === null || value === undefined || value === '';
   272→      case 'select':
   273→        return (value as string) === '' || value === null;
   274→      case 'array':
   275→        return !Array.isArray(value) || value.length === 0;
   276→      case 'ref':
   277→        return value === null || value === undefined;
   278→      default:
   279→        return false;
   280→    }
   281→  };
   282→
   283→  if (isEmpty()) return 'empty';
   284→
   285→  // ── Format validation for non-empty values ──
   286→
   287→  // Number: must be a finite number
   288→  if (field.type === 'number') {
   289→    if (typeof value !== 'number' || !Number.isFinite(value)) return 'invalid';
   290→    return 'ok';
   291→  }
   292→
   293→  // Date: must match YYYY-MM-DD and be a real date
   294→  if (field.type === 'date') {
   295→    const str = String(value);
   296→    if (!/^\d{4}-\d{2}-\d{2}$/.test(str)) return 'invalid';
   297→    const d = new Date(str + 'T00:00:00');
   298→    if (isNaN(d.getTime())) return 'invalid';
   299→    // Verify the parsed date matches the input (e.g. no rollover like 2024-02-30)
   300→    const [y, m, day] = str.split('-').map(Number);
   301→    if (d.getFullYear() !== y || d.getMonth() + 1 !== m || d.getDate() !== day) {
   302→      return 'invalid';
   303→    }
   304→    return 'ok';
   305→  }
   306→
   307→  // Text / textarea: check against format rules by key name
   308→  if (field.type === 'text' || field.type === 'textarea') {
   309→    const str = String(value);
   310→    const rule = findFormatRule(field.key);
   311→    if (rule && !rule.validate(str)) return 'invalid';
   312→    return 'ok';
   313→  }
   314→
   315→  // Select: if value is set, it's always valid (chosen from a list)
   316→  if (field.type === 'select') return 'ok';
   317→
   318→  // Array: if non-empty, valid
   319→  if (field.type === 'array') return 'ok';
   320→
   321→  // Ref: if set, valid
   322→  if (field.type === 'ref') return 'ok';
   323→
   324→  return 'ok';
   325→}

   327→/** Count empty + invalid fields in a record */
   328→export function countFieldIssues(
   329→  fields: FieldDef[],
   330→  row: Record<string, unknown>,
   331→): number {
   332→  let count = 0;
   333→  for (const f of fields) {
   334→    const status = getFieldStatus(f, row[f.key]);
   335→    if (status !== 'ok') count++;
   336→  }
   337→  return count;
   338→}

   604→  const updateNestedCell = useStore((s) => s.updateNestedCell);
   605→  const addNestedRow = useStore((s) => s.addNestedRow);
   606→  const removeNestedRow = useStore((s) => s.removeNestedRow);
   607→  const markDBSourced = useStore((s) => s.markDBSourced);
   608→  const isDBSourced = useStore((s) => s.isDBSourced);
   609→  const allData = useStore((s) => s.data);
   610→
   611→  // Autocomplete state
   612→  const autocompleteResults = useConnectionStore((s) => s.autocompleteResults);
   613→  const autocompleteLoading = useConnectionStore((s) => s.autocompleteLoading);
   614→  const activeAutocompleteField = useConnectionStore((s) => s.activeAutocompleteField);
   615→  const doAutocomplete = useConnectionStore((s) => s.autocomplete);
   616→  const clearAutocomplete = useConnectionStore((s) => s.clearAutocomplete);
   617→  const setActiveAutocompleteField = useConnectionStore((s) => s.setActiveAutocompleteField);
   618→
   619→  const canAutocomplete = isAutocompleteField(field);
   620→  const dbSourced = isDBSourced(sectionKey, rowIndex, field.key);
   621→
   622→  // Ref for click-outside detection
   623→  const dropdownRef = useRef<HTMLDivElement>(null);
   624→
   625→  // Click outside to close dropdown
   626→  useEffect(() => {
   627→    if (!canAutocomplete) return;
   628→    const handler = (e: MouseEvent) => {
   629→      if (dropdownRef.current && !dropdownRef.current.contains(e.target as Node)) {
   630→        clearAutocomplete();
   631→      }
   632→    };
   633→    document.addEventListener('mousedown', handler);
   634→    return () => document.removeEventListener('mousedown', handler);
   635→  }, [canAutocomplete, clearAutocomplete]);
   636→
   637→  // Debounced autocomplete trigger
   638→  const debouncedAutocomplete = useMemo(
   639→    () =>
   640→      debounce((sk: string, fk: string, v: string) => {
   641→        if (v.length >= 2) {
   642→          doAutocomplete(sk, fk, v);
   643→        } else {
   644→          clearAutocomplete();
   645→        }
   646→      }, 400),
   647→    [doAutocomplete, clearAutocomplete],
   648→  );
   649→
   650→  // Autocomplete text change handler
   651→  const handleAutocompleteTextChange = useCallback(
   652→    (v: string) => {
   653→      setActiveAutocompleteField({ sectionKey, fieldKey: field.key, rowIndex });
   654→      debouncedAutocomplete(sectionKey, field.key, v);
   655→    },
   656→    [sectionKey, field.key, rowIndex, setActiveAutocompleteField, debouncedAutocomplete],
   657→  );
   658→
   659→  // Apply a match from the autocomplete dropdown
   660→  const handleApplyMatch = useCallback(
   661→    (row: Record<string, unknown>) => {
   662→      // Get the section's field definitions to match DB columns to form fields
   663→      const sectionDef = SCHEMA_SECTIONS.find((s) => s.key === sectionKey);
   664→      if (!sectionDef) return;
   665→
   666→      for (const [dbKey, dbValue] of Object.entries(row)) {
   667→        // Find a matching field in the section
   668→        const fieldDef = sectionDef.fields.find(
   669→          (f) => f.key === dbKey && !f.readOnly && !f.virtual && f.type === 'text',
   670→        );
   671→        if (!fieldDef) continue;
   672→
   673→        // Set the value
   674→        const strVal = dbValue !== null && dbValue !== undefined ? String(dbValue) : '';
   675→        updateCell(sectionKey, rowIndex, fieldDef.key, strVal);
   676→        markDBSourced(sectionKey, rowIndex, fieldDef.key);
   677→      }
   678→
   679→      clearAutocomplete();
   680→    },
   681→    [sectionKey, rowIndex, allData, updateCell, markDBSourced, clearAutocomplete],
   682→  );
   683→
   684→  const isAutoFilled = field.autoFilled;
   685→  const status = getFieldStatus(field, value);
   686→  const hasIssue = status !== 'ok';
   687→
   688→  // Show the autocomplete dropdown?
   689→  const showAutocomplete =
   690→    canAutocomplete &&
   691→    autocompleteResults.length > 0 &&
   692→    activeAutocompleteField?.sectionKey === sectionKey &&
   693→    activeAutocompleteField?.fieldKey === field.key &&
   694→    activeAutocompleteField?.rowIndex === rowIndex;
   695→
   696→  const labelContent = (
   697→    <div className="flex items-center gap-1.5">
   698→      {hasIssue && (
   699→        <AlertCircle className="size-3.5 text-destructive shrink-0" />
   700→      )}
   701→      <Label
   702→        className={`text-xs font-medium ${isAutoFilled ? 'text-primary' : hasIssue ? 'text-destructive' : 'text-foreground/80'}`}
   703→      >
   704→        {field.label}
   705→      </Label>
   706→      {isAutoFilled && (
   707→        <Badge
   708→          variant="outline"
   709→          className="text-[9px] px-1 py-0 h-3.5 text-primary/70 border-primary/30"
   710→        >
   711→          авто
   712→        </Badge>
   713→      )}
   714→      {dbSourced && !isAutoFilled && (
   715→        <Badge
   716→          variant="outline"
   717→          className="text-[9px] px-1 py-0 h-3.5 text-amber-600 border-amber-400/40 bg-amber-50"
   718→        >
   719→          из БД
   720→        </Badge>
   721→      )}
   722→      {field.hint && (
   723→        <Tooltip>
   724→          <TooltipTrigger asChild>
   725→            <span className="text-muted-foreground/50 cursor-help text-[10px] leading-none">
   726→              ?
   727→            </span>
   728→          </TooltipTrigger>
   729→          <TooltipContent side="top" sideOffset={4} className="max-w-xs">
   730→            <p className="text-xs">{field.hint}</p>
   731→          </TooltipContent>
   732→        </Tooltip>
   733→      )}
   734→    </div>
   735→  );
   736→
   737→  const inputEl = (
   738→    <FieldInput
   739→      field={field}
   740→      value={value}
   741→      sectionKey={sectionKey}
   742→      rowIndex={rowIndex}
   743→      status={status}
   744→      updateCell={updateCell}
   745→      updateNestedCell={updateNestedCell}
   746→      addNestedRow={addNestedRow}
   747→      removeNestedRow={removeNestedRow}
   748→      onTextChange={canAutocomplete ? handleAutocompleteTextChange : undefined}
   749→    />
   750→  );
   751→
   752→  // For object fields (nested tables), render differently
   753→  if (field.type === 'object' && field.nestedFields) {
   754→    return (
   755→      <div className={fullWidth ? '' : 'col-span-2'}>
   756→        <div className="flex items-center gap-2 mb-2">{labelContent}</div>
   757→        <NestedFieldBlock
   758→          field={field}
   759→          value={value}
   760→          sectionKey={sectionKey}
   761→          rowIndex={rowIndex}
   762→          updateNestedCell={updateNestedCell}
   763→          addNestedRow={addNestedRow}
   764→          removeNestedRow={removeNestedRow}
   765→        />
   766→      </div>
   767→    );
   768→  }
   769→
   770→  // For array fields that need more space
   771→  if (field.type === 'array' && fullWidth) {
   772→    return (
   773→      <div>
   774→        {labelContent}
   775→        <div className="mt-1">{inputEl}</div>
   776→      </div>
   777→    );
   778→  }
   779→
   780→  // For textarea in grid, span 2 columns
   781→  if (field.type === 'textarea') {
   782→    return (
   783→      <div className="col-span-2">
   784→        {labelContent}
   785→        <div className="mt-1">{inputEl}</div>
   786→      </div>
   787→    );
   788→  }
   789→
   790→  // Text fields with autocomplete — wrap in relative container for dropdown
   791→  if (canAutocomplete) {
   792→    return (
   793→      <div>
   794→        {labelContent}
   795→        <div className="mt-1 relative" ref={dropdownRef}>
   796→          {inputEl}
   797→          {autocompleteLoading &&
   798→            activeAutocompleteField?.sectionKey === sectionKey &&
   799→            activeAutocompleteField?.fieldKey === field.key &&
   800→            activeAutocompleteField?.rowIndex === rowIndex && (
   801→              <div className="absolute z-50 mt-1 w-full bg-popover border rounded-md shadow-lg p-2 flex items-center justify-center">
   802→                <Loader2 className="size-3.5 animate-spin text-muted-foreground" />
   803→                <span className="ml-2 text-xs text-muted-foreground">Поиск…</span>
   804→              </div>
   805→            )}
   806→          {showAutocomplete && (
   807→            <div className="absolute z-50 mt-1 w-full bg-popover border rounded-md shadow-lg max-h-60 overflow-y-auto">
   808→              {autocompleteResults.map((match, idx) => (
   809→                <button
   810→                  key={idx}
   811→                  type="button"
   812→                  className="w-full text-left px-3 py-2 text-sm hover:bg-accent transition-colors border-b last:border-b-0"
   813→                  onClick={() => handleApplyMatch(match.row)}
   814→                >
   815→                  <div className="font-medium">{match.label}</div>
   816→                  <div className="text-xs text-muted-foreground">
   817→                    {Object.entries(match.row)
   818→                      .slice(0, 3)
   819→                      .map(([k, v]) => `${k}: ${v}`)
   820→                      .join(' · ')}
   821→                  </div>
   822→                </button>
   823→              ))}
   824→            </div>
   825→          )}
   826→        </div>
   827→      </div>
   828→    );
   829→  }
   830→
   831→  return (
   832→    <div>
   833→      {labelContent}
   834→      <div className="mt-1">{inputEl}</div>
   835→    </div>
   836→  );
   837→}

   876→interface FieldInputProps {

   878→  value: unknown;
   879→  sectionKey: string;
   880→  rowIndex: number;
   881→  status: FieldStatus;
   882→  updateCell: (
   883→    sectionKey: string,
   884→    rowIndex: number,
   885→    fieldKey: string,
   886→    value: unknown,
   887→  ) => void;
   888→  updateNestedCell: (
   889→    sectionKey: string,
   890→    rowIndex: number,
   891→    fieldKey: string,
   892→    nestedIndex: number,

   896→  addNestedRow: (

   903→    rowIndex: number,

   905→    nestedIndex: number,

   907→  /** Optional extra onChange handler for text inputs (used by autocomplete) */
   908→  onTextChange?: (value: string) => void;

   910→
   911→function FieldInput({
   912→  field,
   913→  value,
   914→  sectionKey,
   915→  rowIndex,
   916→  status,
   917→  updateCell,
   918→  onTextChange,
   919→}: FieldInputProps) {
   920→  const isAutoFilled = field.autoFilled;
   921→  const hasIssue = status !== 'ok';
   922→  const isInvalid = status === 'invalid';
   923→
   924→  // Build class based on status
   925→  const baseInputClass = isAutoFilled
   926→    ? 'h-9 text-sm border-primary/25 bg-primary/5 focus-visible:ring-primary/40'
   927→    : hasIssue
   928→      ? 'h-9 text-sm border-destructive/50 bg-destructive/5 focus-visible:ring-destructive/40'
   929→      : 'h-9 text-sm focus-visible:ring-1 focus-visible:ring-ring';
   930→
   931→  switch (field.type) {
   932→    case 'text': {
   933→      const inputEl = (
   934→        <Input
   935→          type="text"
   936→          value={(value as string) || ''}
   937→          placeholder={field.placeholder}
   938→          className={baseInputClass}
   939→          onChange={(e) => {
   940→            updateCell(sectionKey, rowIndex, field.key, e.target.value);
   941→            onTextChange?.(e.target.value);
   942→          }}
   943→        />
   944→      );
   945→      return isInvalid ? (
   946→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   947→      ) : (
   948→        inputEl
   949→      );
   950→    }
   951→
   952→    case 'textarea': {
   953→      const taClass = hasIssue
   954→        ? 'min-h-[80px] text-sm resize-y border-destructive/50 bg-destructive/5 focus-visible:ring-destructive/40'
   955→        : 'min-h-[80px] text-sm resize-y focus-visible:ring-1 focus-visible:ring-ring';
   956→      const inputEl = (
   957→        <Textarea
   958→          value={(value as string) || ''}
   959→          placeholder={field.placeholder}
   960→          rows={3}
   961→          className={taClass}
   962→          onChange={(e) =>
   963→            updateCell(sectionKey, rowIndex, field.key, e.target.value)
   964→          }
   965→        />
   966→      );
   967→      return isInvalid ? (
   968→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   969→      ) : (
   970→        inputEl
   971→      );
   972→    }
   973→
   974→    case 'number': {
   975→      const inputEl = (
   976→        <Input
   977→          type="number"
   978→          step="any"
   979→          value={
   980→            value === null || value === undefined ? '' : String(value)
   981→          }
   982→          placeholder={field.placeholder}
   983→          className={baseInputClass}
   984→          onChange={(e) => {
   985→            const v = e.target.value;
   986→            updateCell(
   987→              sectionKey,
   988→              rowIndex,
   989→              field.key,
   990→              v === '' ? null : Number(v),
   991→            );
   992→          }}
   993→        />
   994→      );
   995→      return isInvalid ? (
   996→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   997→      ) : (
   998→        inputEl
   999→      );
  1000→    }
  1001→
  1002→    case 'date': {
  1003→      const inputEl = (
  1004→        <Input
  1005→          type="date"
  1006→          value={(value as string) || ''}
  1007→          className={baseInputClass}
  1008→          onChange={(e) =>
  1009→            updateCell(
  1010→              sectionKey,
  1011→              rowIndex,
  1012→              field.key,
  1013→              e.target.value || null,
  1014→            )
  1015→          }
  1016→        />
  1017→      );
  1018→      return isInvalid ? (
  1019→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
  1020→      ) : (
  1021→        inputEl
  1022→      );
  1023→    }
  1024→
  1025→    case 'boolean':
  1026→      return (
  1027→        <div className="flex items-center h-9">
  1028→          <Switch
  1029→            checked={!!value}
  1030→            onCheckedChange={(checked) =>
  1031→              updateCell(sectionKey, rowIndex, field.key, checked)
  1032→            }
  1033→          />
  1034→        </div>
  1035→      );
  1036→
  1037→    case 'select': {
  1038→      const selectEl = (
  1039→        <Select
  1040→          value={(value as string) || ''}
  1041→          onValueChange={(v) =>
  1042→            updateCell(sectionKey, rowIndex, field.key, v)
  1043→          }
  1044→        >
  1045→          <SelectTrigger
  1046→            className={
  1047→              isAutoFilled
  1048→                ? 'h-9 text-sm border-primary/25 bg-primary/5 focus:ring-primary/40 w-full'
  1049→                : hasIssue
  1050→                  ? 'h-9 text-sm border-destructive/50 bg-destructive/5 focus:ring-destructive/40 w-full'
  1051→                  : 'h-9 text-sm w-full'
  1052→            }
  1053→          >
  1054→            <SelectValue placeholder={field.placeholder || 'Выберите…'} />
  1055→          </SelectTrigger>
  1056→          <SelectContent>
  1057→            {(field.options || []).map((opt) => (
  1058→              <SelectItem key={opt} value={opt}>
  1059→                {opt}
  1060→              </SelectItem>
  1061→            ))}
  1062→          </SelectContent>
  1063→        </Select>
  1064→      );
  1065→      return isInvalid ? (
  1066→        <InvalidFieldTooltip field={field}>{selectEl}</InvalidFieldTooltip>
  1067→      ) : (
  1068→        selectEl
  1069→      );
  1070→    }
  1071→
  1072→    case 'array': {
  1073→      const arr = Array.isArray(value) ? (value as string[]) : [];
  1074→      const inputEl = (
  1075→        <Input
  1076→          type="text"
  1077→          value={arr.join(', ')}
  1078→          placeholder="через запятую"
  1079→          className={baseInputClass}
  1080→          onChange={(e) => {
  1081→            const items = e.target.value
  1082→              .split(',')
  1083→              .map((s) => s.trim())
  1084→              .filter(Boolean);
  1085→            updateCell(sectionKey, rowIndex, field.key, items);
  1086→          }}
  1087→        />
  1088→      );
  1089→      return isInvalid ? (
  1090→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
  1091→      ) : (
  1092→        inputEl
  1093→      );
  1094→    }
  1095→
  1096→    case 'ref': {
  1097→      const refEl = (
  1098→        <RefSelect
  1099→          field={field}
  1100→          value={value}
  1101→          sectionKey={sectionKey}
  1102→          rowIndex={rowIndex}
  1103→          hasIssue={hasIssue}
  1104→        />
  1105→      );
  1106→      return isInvalid ? (
  1107→        <InvalidFieldTooltip field={field}>{refEl}</InvalidFieldTooltip>
  1108→      ) : (
  1109→        refEl
  1110→      );
  1111→    }
  1112→
  1113→    default:
  1114→      return <span className="text-muted-foreground text-xs">—</span>;
  1115→  }
  1116→}

  1119→// RefSelect — dropdown populated from another section's data
  1120→// =============================================================================
  1121→
  1122→function RefSelect({
  1123→  field,
  1124→  value,
  1125→  sectionKey,
  1126→  rowIndex,
  1127→  hasIssue,
  1128→}: {
  1129→  field: FieldDef;
  1130→  value: unknown;
  1131→  sectionKey: string;
  1132→  rowIndex: number;
  1133→  hasIssue: boolean;
  1134→}) {
  1135→  const updateCell = useStore((s) => s.updateCell);
  1136→  const allData = useStore((s) => s.data);
  1137→  const refSection = field.refSection;
  1138→  const refLabelField = field.refLabelField;
  1139→
  1140→  const options = useMemo(() => {
  1141→    if (!refSection) return [];
  1142→    const rows = (allData[refSection] || []) as Record<string, unknown>[];
  1143→    return rows
  1144→      .map((row, idx) => {
  1145→        const label = refLabelField
  1146→          ? String(row[refLabelField] || '')
  1147→          : `Запись ${idx + 1}`;
  1148→        return { idx, label };
  1149→      })
  1150→      .filter((o) => o.label);
  1151→  }, [allData, refSection, refLabelField]);
  1152→
  1153→  const currentIdx = value as number | null;
  1154→  const currentValue =
  1155→    currentIdx !== null && currentIdx !== undefined
  1156→      ? String(currentIdx)
  1157→      : '';
  1158→  const currentLabel =
  1159→    currentIdx !== null && currentIdx !== undefined
  1160→      ? options.find((o) => o.idx === currentIdx)?.label || ''
  1161→      : '';
  1162→
  1163→  const isVirtual = field.virtual;
  1164→  const triggerClass = hasIssue
  1165→    ? 'h-9 text-sm w-full border-destructive/50 bg-destructive/5 focus:ring-destructive/40'
  1166→    : isVirtual
  1167→      ? 'h-9 text-sm w-full border-primary/30 bg-primary/5 focus:ring-1 focus:ring-primary'
  1168→      : 'h-9 text-sm w-full';
  1169→
  1170→  return (
  1171→    <Select
  1172→      value={currentValue}
  1173→      onValueChange={(v) =>
  1174→        updateCell(sectionKey, rowIndex, field.key, Number(v))
  1175→      }
  1176→    >
  1177→      <SelectTrigger className={triggerClass}>
  1178→        <SelectValue placeholder={field.placeholder || 'Выберите…'}>
  1179→          {currentLabel || (field.placeholder || 'Выберите…')}
  1180→        </SelectValue>
  1181→      </SelectTrigger>
  1182→      <SelectContent>
  1183→        {options.length === 0 && (
  1184→          <SelectItem value="_empty" disabled>
  1185→            Нет записей в справочнике
  1186→          </SelectItem>
  1187→        )}
  1188→        {options.map((opt) => (
  1189→          <SelectItem key={opt.idx} value={String(opt.idx)}>
  1190→            {opt.label}
  1191→          </SelectItem>
  1192→        ))}
  1193→      </SelectContent>
  1194→    </Select>
  1195→  );
  1196→}

  1380→function NestedFieldInput({
  1381→  nestedField,
  1382→  value,
  1383→  onChange,
  1384→}: {
  1385→  nestedField: FieldDef;
  1386→  value: unknown;
  1387→  onChange: (value: unknown) => void;
  1388→}) {
  1389→  const v = value ?? '';
  1390→
  1391→  switch (nestedField.type) {
  1392→    case 'boolean':
  1393→      return (
  1394→        <Switch
  1395→          checked={!!value}
  1396→          onCheckedChange={(checked) => onChange(checked)}
  1397→          className="scale-75"
  1398→        />
  1399→      );
  1400→    case 'number':
  1401→      return (
  1402→        <input
  1403→          type="number"
  1404→          step="any"
  1405→          value={v === '' || v === null ? '' : String(v)}
  1406→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
  1407→          onChange={(e) => {
  1408→            const nv = e.target.value;
  1409→            onChange(nv === '' ? null : Number(nv));
  1410→          }}
  1411→        />
  1412→      );
  1413→    case 'select':
  1414→      return (
  1415→        <select
  1416→          value={String(v)}
  1417→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
  1418→          onChange={(e) => onChange(e.target.value)}
  1419→        >
  1420→          <option value="">—</option>
  1421→          {(nestedField.options || []).map((opt) => (
  1422→            <option key={opt} value={opt}>
  1423→              {opt}
  1424→            </option>
  1425→          ))}
  1426→        </select>
  1427→      );
  1428→    default:
  1429→      return (
  1430→        <input
  1431→          type="text"
  1432→          value={String(v)}
  1433→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
  1434→          onChange={(e) => onChange(e.target.value)}
  1435→        />
  1436→      );
  1437→  }
  1438→}
