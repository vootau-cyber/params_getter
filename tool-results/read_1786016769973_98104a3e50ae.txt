  4303→export function getEmptyRow(sectionKey: string): Record<string, unknown> {
  4304→  const section = SCHEMA_SECTIONS.find((s) => s.key === sectionKey);
  4305→  if (!section) return {};
  4306→
  4307→  const row: Record<string, unknown> = {};
  4308→  for (const field of section.fields) {
  4309→    if (field.type === "object" && field.nestedFields) {
  4310→      if (Array.isArray(field.defaultValue)) {
  4311→        row[field.key] = field.defaultValue;
  4312→      } else {
  4313→        row[field.key] = {};
  4314→      }
  4315→    } else if (field.type === "array") {
  4316→      row[field.key] = [];
  4317→    } else if (field.type === "boolean") {
  4318→      row[field.key] = field.defaultValue ?? false;
  4319→    } else if (field.type === "number") {
  4320→      row[field.key] = field.defaultValue ?? null;
  4321→    } else if (field.type === "date") {
  4322→      row[field.key] = field.defaultValue ?? null;
  4323→    } else if (field.type === "ref") {
  4324→      row[field.key] = field.defaultValue ?? null;
  4325→    } else {
  4326→      row[field.key] = field.defaultValue ?? "";
  4327→    }
  4328→  }
  4329→  return row;
  4330→}
