   182→export type FieldStatus = 'empty' | 'invalid' | 'ok';

   184→/**
   185→ * Find the matching format rule for a field key, or null if none.
   186→ */
   187→function findFormatRule(fieldKey: string): FormatRule | null {
   188→  for (const rule of FORMAT_RULES) {
   189→    if (rule.keyPattern.test(fieldKey)) return rule;
   190→  }
   191→  return null;
   192→}

   194→/**
   195→ * Get the Russian format hint for a field (for tooltips).
   196→ * Returns null if the field has no specific format requirement.
   197→ */
   198→export function getFieldFormatHint(field: FieldDef): string | null {
   199→  // Number fields: always validate as finite number
   200→  if (field.type === 'number') {
   201→    return 'Число (дробное через точку)';
   202→  }
   203→  // Date fields: always validate as YYYY-MM-DD
   204→  if (field.type === 'date') {
   205→    return 'ГГГГ-ММ-ДД (например, 2024-12-31)';
   206→  }
   207→  // Text/textarea: check against FORMAT_RULES by key name
   208→  const rule = findFormatRule(field.key);
   209→  return rule ? rule.formatHint : null;
   210→}
   211→
   212→/**
   213→ * Determine if a field value is "empty" or "invalid".
   214→ * Booleans, readOnly, virtual, and object fields are always 'ok'.
   215→ *
   216→ * Returns:
   217→ *   'empty'   — value is missing / unfilled
   218→ *   'invalid' — value is present but fails format validation
   219→ *   'ok'      — value is present and valid
   220→ */
   221→export function getFieldStatus(field: FieldDef, value: unknown): FieldStatus {
   222→  // Skip types that don't need validation
   223→  if (field.readOnly || field.virtual) return 'ok';
   224→  if (field.type === 'boolean') return 'ok';
   225→  if (field.type === 'object') return 'ok';
   226→
   227→  // Helper: is the value considered "empty" for this type?
   228→  const isEmpty = (): boolean => {
   229→    switch (field.type) {
   230→      case 'text':
   231→      case 'textarea':
   232→        return (value as string) === '' || value === undefined || value === null;
   233→      case 'number':
   234→        return value === null || value === undefined;
   235→      case 'date':
   236→        return value === null || value === undefined || value === '';
   237→      case 'select':
   238→        return (value as string) === '' || value === null;
   239→      case 'array':
   240→        return !Array.isArray(value) || value.length === 0;
   241→      case 'ref':
   242→        return value === null || value === undefined;
   243→      default:
   244→        return false;
   245→    }
   246→  };
   247→
   248→  if (isEmpty()) return 'empty';
   249→
   250→  // ── Format validation for non-empty values ──
   251→
   252→  // Number: must be a finite number
   253→  if (field.type === 'number') {
   254→    if (typeof value !== 'number' || !Number.isFinite(value)) return 'invalid';
   255→    return 'ok';
   256→  }
   257→
   258→  // Date: must match YYYY-MM-DD and be a real date
   259→  if (field.type === 'date') {
   260→    const str = String(value);
   261→    if (!/^\d{4}-\d{2}-\d{2}$/.test(str)) return 'invalid';
   262→    const d = new Date(str + 'T00:00:00');
   263→    if (isNaN(d.getTime())) return 'invalid';
   264→    // Verify the parsed date matches the input (e.g. no rollover like 2024-02-30)
   265→    const [y, m, day] = str.split('-').map(Number);
   266→    if (d.getFullYear() !== y || d.getMonth() + 1 !== m || d.getDate() !== day) {
   267→      return 'invalid';
   268→    }
   269→    return 'ok';
   270→  }
   271→
   272→  // Text / textarea: check against format rules by key name
   273→  if (field.type === 'text' || field.type === 'textarea') {
   274→    const str = String(value);
   275→    const rule = findFormatRule(field.key);
   276→    if (rule && !rule.validate(str)) return 'invalid';
   277→    return 'ok';
   278→  }
   279→
   280→  // Select: if value is set, it's always valid (chosen from a list)
   281→  if (field.type === 'select') return 'ok';
   282→
   283→  // Array: if non-empty, valid
   284→  if (field.type === 'array') return 'ok';
   285→
   286→  // Ref: if set, valid
   287→  if (field.type === 'ref') return 'ok';
   288→
   289→  return 'ok';
   290→}

   292→/** Count empty + invalid fields in a record */
   293→export function countFieldIssues(
   294→  fields: FieldDef[],
   295→  row: Record<string, unknown>,
   296→): number {
   297→  let count = 0;
   298→  for (const f of fields) {
   299→    const status = getFieldStatus(f, row[f.key]);
   300→    if (status !== 'ok') count++;
   301→  }
   302→  return count;
   303→}

   646→        {labelContent}
   647→        <div className="mt-1">{inputEl}</div>
   648→      </div>
   649→    );

   660→  }
   661→
   662→  return (
   663→    <div>
   664→      {labelContent}
   665→      <div className="mt-1">{inputEl}</div>
   666→    </div>
   667→  );
   668→}

   740→function FieldInput({
   741→  field,
   742→  value,
   743→  sectionKey,
   744→  rowIndex,
   745→  status,
   746→  updateCell,
   747→}: FieldInputProps) {
   748→  const isAutoFilled = field.autoFilled;
   749→  const hasIssue = status !== 'ok';
   750→  const isInvalid = status === 'invalid';
   751→
   752→  // Build class based on status
   753→  const baseInputClass = isAutoFilled
   754→    ? 'h-9 text-sm border-primary/25 bg-primary/5 focus-visible:ring-primary/40'
   755→    : hasIssue
   756→      ? 'h-9 text-sm border-destructive/50 bg-destructive/5 focus-visible:ring-destructive/40'
   757→      : 'h-9 text-sm focus-visible:ring-1 focus-visible:ring-ring';
   758→
   759→  switch (field.type) {
   760→    case 'text': {
   761→      const inputEl = (
   762→        <Input
   763→          type="text"
   764→          value={(value as string) || ''}
   765→          placeholder={field.placeholder}
   766→          className={baseInputClass}
   767→          onChange={(e) =>
   768→            updateCell(sectionKey, rowIndex, field.key, e.target.value)
   769→          }
   770→        />
   771→      );
   772→      return isInvalid ? (
   773→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   774→      ) : (
   775→        inputEl
   776→      );
   777→    }
   778→
   779→    case 'textarea': {
   780→      const taClass = hasIssue
   781→        ? 'min-h-[80px] text-sm resize-y border-destructive/50 bg-destructive/5 focus-visible:ring-destructive/40'
   782→        : 'min-h-[80px] text-sm resize-y focus-visible:ring-1 focus-visible:ring-ring';
   783→      const inputEl = (
   784→        <Textarea
   785→          value={(value as string) || ''}
   786→          placeholder={field.placeholder}
   787→          rows={3}
   788→          className={taClass}
   789→          onChange={(e) =>
   790→            updateCell(sectionKey, rowIndex, field.key, e.target.value)
   791→          }
   792→        />
   793→      );
   794→      return isInvalid ? (
   795→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   796→      ) : (
   797→        inputEl
   798→      );
   799→    }
   800→
   801→    case 'number': {
   802→      const inputEl = (
   803→        <Input
   804→          type="number"
   805→          step="any"
   806→          value={
   807→            value === null || value === undefined ? '' : String(value)
   808→          }
   809→          placeholder={field.placeholder}
   810→          className={baseInputClass}
   811→          onChange={(e) => {
   812→            const v = e.target.value;
   813→            updateCell(
   814→              sectionKey,
   815→              rowIndex,
   816→              field.key,
   817→              v === '' ? null : Number(v),
   818→            );
   819→          }}
   820→        />
   821→      );
   822→      return isInvalid ? (
   823→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   824→      ) : (
   825→        inputEl
   826→      );
   827→    }
   828→
   829→    case 'date': {
   830→      const inputEl = (
   831→        <Input
   832→          type="date"
   833→          value={(value as string) || ''}
   834→          className={baseInputClass}
   835→          onChange={(e) =>
   836→            updateCell(
   837→              sectionKey,
   838→              rowIndex,
   839→              field.key,
   840→              e.target.value || null,
   841→            )
   842→          }
   843→        />
   844→      );
   845→      return isInvalid ? (
   846→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   847→      ) : (
   848→        inputEl
   849→      );
   850→    }
   851→
   852→    case 'boolean':
   853→      return (
   854→        <div className="flex items-center h-9">
   855→          <Switch
   856→            checked={!!value}
   857→            onCheckedChange={(checked) =>
   858→              updateCell(sectionKey, rowIndex, field.key, checked)
   859→            }
   860→          />
   861→        </div>
   862→      );
   863→
   864→    case 'select': {
   865→      const selectEl = (
   866→        <Select
   867→          value={(value as string) || ''}
   868→          onValueChange={(v) =>
   869→            updateCell(sectionKey, rowIndex, field.key, v)
   870→          }
   871→        >
   872→          <SelectTrigger
   873→            className={
   874→              isAutoFilled
   875→                ? 'h-9 text-sm border-primary/25 bg-primary/5 focus:ring-primary/40 w-full'
   876→                : hasIssue
   877→                  ? 'h-9 text-sm border-destructive/50 bg-destructive/5 focus:ring-destructive/40 w-full'
   878→                  : 'h-9 text-sm w-full'
   879→            }
   880→          >
   881→            <SelectValue placeholder={field.placeholder || 'Выберите…'} />
   882→          </SelectTrigger>
   883→          <SelectContent>
   884→            {(field.options || []).map((opt) => (
   885→              <SelectItem key={opt} value={opt}>
   886→                {opt}
   887→              </SelectItem>
   888→            ))}
   889→          </SelectContent>
   890→        </Select>
   891→      );
   892→      return isInvalid ? (
   893→        <InvalidFieldTooltip field={field}>{selectEl}</InvalidFieldTooltip>
   894→      ) : (
   895→        selectEl
   896→      );
   897→    }
   898→
   899→    case 'array': {
   900→      const arr = Array.isArray(value) ? (value as string[]) : [];
   901→      const inputEl = (
   902→        <Input
   903→          type="text"
   904→          value={arr.join(', ')}
   905→          placeholder="через запятую"
   906→          className={baseInputClass}
   907→          onChange={(e) => {
   908→            const items = e.target.value
   909→              .split(',')
   910→              .map((s) => s.trim())
   911→              .filter(Boolean);
   912→            updateCell(sectionKey, rowIndex, field.key, items);
   913→          }}
   914→        />
   915→      );
   916→      return isInvalid ? (
   917→        <InvalidFieldTooltip field={field}>{inputEl}</InvalidFieldTooltip>
   918→      ) : (
   919→        inputEl
   920→      );
   921→    }
   922→
   923→    case 'ref': {
   924→      const refEl = (
   925→        <RefSelect
   926→          field={field}
   927→          value={value}
   928→          sectionKey={sectionKey}
   929→          rowIndex={rowIndex}
   930→          hasIssue={hasIssue}
   931→        />
   932→      );
   933→      return isInvalid ? (
   934→        <InvalidFieldTooltip field={field}>{refEl}</InvalidFieldTooltip>
   935→      ) : (
   936→        refEl
   937→      );
   938→    }
   939→
   940→    default:
   941→      return <span className="text-muted-foreground text-xs">—</span>;
   942→  }
   943→}

   949→function RefSelect({
   950→  field,
   951→  value,
   952→  sectionKey,
   953→  rowIndex,
   954→  hasIssue,
   955→}: {
   956→  field: FieldDef;
   957→  value: unknown;
   958→  sectionKey: string;
   959→  rowIndex: number;
   960→  hasIssue: boolean;
   961→}) {
   962→  const updateCell = useStore((s) => s.updateCell);
   963→  const allData = useStore((s) => s.data);
   964→  const refSection = field.refSection;
   965→  const refLabelField = field.refLabelField;
   966→
   967→  const options = useMemo(() => {
   968→    if (!refSection) return [];
   969→    const rows = (allData[refSection] || []) as Record<string, unknown>[];
   970→    return rows
   971→      .map((row, idx) => {
   972→        const label = refLabelField
   973→          ? String(row[refLabelField] || '')
   974→          : `Запись ${idx + 1}`;
   975→        return { idx, label };
   976→      })
   977→      .filter((o) => o.label);
   978→  }, [allData, refSection, refLabelField]);
   979→
   980→  const currentIdx = value as number | null;
   981→  const currentValue =
   982→    currentIdx !== null && currentIdx !== undefined
   983→      ? String(currentIdx)
   984→      : '';
   985→  const currentLabel =
   986→    currentIdx !== null && currentIdx !== undefined
   987→      ? options.find((o) => o.idx === currentIdx)?.label || ''
   988→      : '';
   989→
   990→  const isVirtual = field.virtual;
   991→  const triggerClass = hasIssue
   992→    ? 'h-9 text-sm w-full border-destructive/50 bg-destructive/5 focus:ring-destructive/40'
   993→    : isVirtual
   994→      ? 'h-9 text-sm w-full border-primary/30 bg-primary/5 focus:ring-1 focus:ring-primary'
   995→      : 'h-9 text-sm w-full';
   996→
   997→  return (
   998→    <Select
   999→      value={currentValue}
  1000→      onValueChange={(v) =>
  1001→        updateCell(sectionKey, rowIndex, field.key, Number(v))
  1002→      }
  1003→    >
  1004→      <SelectTrigger className={triggerClass}>
  1005→        <SelectValue placeholder={field.placeholder || 'Выберите…'}>
  1006→          {currentLabel || (field.placeholder || 'Выберите…')}
  1007→        </SelectValue>
  1008→      </SelectTrigger>
  1009→      <SelectContent>
  1010→        {options.length === 0 && (
  1011→          <SelectItem value="_empty" disabled>
  1012→            Нет записей в справочнике
  1013→          </SelectItem>
  1014→        )}
  1015→        {options.map((opt) => (
  1016→          <SelectItem key={opt.idx} value={String(opt.idx)}>
  1017→            {opt.label}
  1018→          </SelectItem>
  1019→        ))}
  1020→      </SelectContent>
  1021→    </Select>
  1022→  );
  1023→}

  1138→                        removeNestedRow(
  1139→                          sectionKey,
  1140→                          rowIndex,
  1141→                          field.key,

  1143→                        )

  1171→
  1172→  // Single nested object (e.g., ztb_boundaries)

  1179→            <Label className="text-xs font-medium text-foreground/80">

  1199→
  1200→  return null;
  1201→}

  1207→function NestedFieldInput({
  1208→  nestedField,
  1209→  value,
  1210→  onChange,
  1211→}: {
  1212→  nestedField: FieldDef;
  1213→  value: unknown;
  1214→  onChange: (value: unknown) => void;
  1215→}) {
  1216→  const v = value ?? '';
  1217→
  1218→  switch (nestedField.type) {
  1219→    case 'boolean':

  1223→          onCheckedChange={(checked) => onChange(checked)}

  1226→      );

  1228→      return (
  1229→        <input
  1230→          type="number"
  1231→          step="any"
  1232→          value={v === '' || v === null ? '' : String(v)}
  1233→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
  1234→          onChange={(e) => {
  1235→            const nv = e.target.value;
  1236→            onChange(nv === '' ? null : Number(nv));
  1237→          }}
  1238→        />
  1239→      );
  1240→    case 'select':
  1241→      return (
  1242→        <select
  1243→          value={String(v)}
  1244→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
  1245→          onChange={(e) => onChange(e.target.value)}
  1246→        >
  1247→          <option value="">—</option>
  1248→          {(nestedField.options || []).map((opt) => (
  1249→            <option key={opt} value={opt}>
  1250→              {opt}
  1251→            </option>
  1252→          ))}
  1253→        </select>
  1254→      );
  1255→    default:
  1256→      return (
  1257→        <input
  1258→          type="text"
  1259→          value={String(v)}
  1260→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
  1261→          onChange={(e) => onChange(e.target.value)}
  1262→        />
  1263→      );
  1264→  }
  1265→}
