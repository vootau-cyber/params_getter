  4217→export function getEmptyRow(sectionKey: string): Record<string, unknown> {
  4218→  const section = SCHEMA_SECTIONS.find((s) => s.key === sectionKey);
  4219→  if (!section) return {};
  4220→
  4221→  const row: Record<string, unknown> = {};
  4222→  for (const field of section.fields) {
  4223→    if (field.type === "object" && field.nestedFields) {
  4224→      if (Array.isArray(field.defaultValue)) {
  4225→        row[field.key] = field.defaultValue;
  4226→      } else {
  4227→        row[field.key] = {};
  4228→      }
  4229→    } else if (field.type === "array") {
  4230→      row[field.key] = [];
  4231→    } else if (field.type === "boolean") {
  4232→      row[field.key] = field.defaultValue ?? false;
  4233→    } else if (field.type === "number") {
  4234→      row[field.key] = field.defaultValue ?? null;
  4235→    } else if (field.type === "date") {
  4236→      row[field.key] = field.defaultValue ?? null;
  4237→    } else if (field.type === "ref") {
  4238→      row[field.key] = field.defaultValue ?? null;
  4239→    } else {
  4240→      row[field.key] = field.defaultValue ?? "";
  4241→    }
  4242→  }
  4243→  return row;
  4244→}
