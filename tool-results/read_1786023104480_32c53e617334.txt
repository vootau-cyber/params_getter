    49→
    50→export function SectionTable({ section }: SectionTableProps) {
    51→  const data = useStore((s) => s.data[section.key] || []);
    52→  const addRow = useStore((s) => s.addRow);
    53→  const removeRow = useStore((s) => s.removeRow);
    54→  const [activeRecord, setActiveRecord] = useState(0);
    55→
    56→  // Clamp activeRecord when data changes
    57→  const clampedRecord = data.length > 0 ? Math.min(activeRecord, data.length - 1) : 0;
    58→
    59→  // If activeRecord was clamped, update
    60→  React.useEffect(() => {
    61→    if (clampedRecord !== activeRecord) {
    62→      setActiveRecord(clampedRecord);
    63→    }
    64→  }, [clampedRecord, activeRecord]);
    65→
    66→  const editableFields = section.fields.filter((f) => !f.readOnly);
    67→  const readOnlyFields = section.fields.filter((f) => f.readOnly);
    68→  const row = data.length > 0 ? (data[clampedRecord] as Record<string, unknown>) : null;
    69→
    70→  // Empty state
    71→  if (data.length === 0) {
    72→    return (
    73→      <div className="py-16 text-center">
    74→        <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-muted">
    75→          <FileText className="size-7 text-muted-foreground" />
    76→        </div>
    77→        <p className="text-muted-foreground text-sm mb-4">
    78→          Нет данных. Добавьте запись для начала.
    79→        </p>
    80→        <Button
    81→          variant="outline"
    82→          onClick={() => {
    83→            addRow(section.key);
    84→            setActiveRecord(0);
    85→          }}
    86→        >
    87→          <Plus className="mr-2 size-4" />
    88→          Добавить запись
    89→        </Button>
    90→      </div>
    91→    );
    92→  }
    93→
    94→  // Classify fields for layout
    95→  const shortFields = editableFields.filter(
    96→    (f) =>
    97→      !f.virtual &&
    98→      f.type !== 'textarea' &&
    99→      f.type !== 'object' &&
   100→      f.type !== 'array',
   101→  );
   102→  const virtualFields = editableFields.filter((f) => f.virtual);
   103→  const textareaFields = editableFields.filter((f) => f.type === 'textarea');
   104→  const arrayFields = editableFields.filter((f) => f.type === 'array');
   105→  const objectFields = editableFields.filter((f) => f.type === 'object');
   106→
   107→  // Split short fields into pairs for 2-column grid
   108→  const fieldPairs: (FieldDef | null)[][] = [];
   109→  for (let i = 0; i < shortFields.length; i += 2) {
   110→    fieldPairs.push([shortFields[i], shortFields[i + 1] ?? null]);
   111→  }
   112→
   113→  return (
   114→    <div className="space-y-4">
   115→      {/* ── Record Tabs ─────────────────────────────────────────────── */}
   116→      <div className="flex items-center gap-2 border-b pb-3">
   117→        <div className="flex items-center gap-1 overflow-x-auto">
   118→          {data.map((_, idx) => (
   119→            <Button
   120→              key={idx}
   121→              variant={idx === clampedRecord ? 'default' : 'outline'}
   122→              size="sm"
   123→              className="shrink-0 h-8 px-3 text-xs"
   124→              onClick={() => setActiveRecord(idx)}
   125→            >
   126→              Запись {idx + 1}
   127→            </Button>
   128→          ))}
   129→        </div>
   130→        <Button
   131→          variant="outline"
   132→          size="sm"
   133→          className="shrink-0 h-8 text-xs"
   134→          onClick={() => {
   135→            addRow(section.key);
   136→            // New record is at the end, switch to it
   137→            setTimeout(() => setActiveRecord(data.length), 0);
   138→          }}
   139→        >
   140→          <Plus className="mr-1 size-3.5" />
   141→          Добавить
   142→        </Button>
   143→        <div className="ml-auto">
   144→          <Button
   145→            variant="outline"
   146→            size="sm"
   147→            className="h-8 text-xs text-destructive hover:text-destructive"
   148→            onClick={() => {
   149→              removeRow(section.key, clampedRecord);
   150→              if (clampedRecord >= data.length - 1 && clampedRecord > 0) {
   151→                setActiveRecord(clampedRecord - 1);
   152→              }
   153→            }}
   154→          >
   155→            <Trash2 className="mr-1 size-3.5" />
   156→            Удалить
   157→          </Button>
   158→        </div>
   159→      </div>
   160→
   161→      {/* ── Record Form ──────────────────────────────────────────────── */}
   162→      {row && (
   163→        <div className="space-y-6">
   164→          {/* Virtual ref fields (reference selectors) */}
   165→          {virtualFields.length > 0 && (
   166→            <div className="rounded-lg border border-primary/20 bg-primary/5 p-4 space-y-3">
   167→              <div className="flex items-center gap-2 text-xs font-semibold text-primary uppercase tracking-wider">
   168→                <Link2 className="size-3.5" />
   169→                Ссылки на другие разделы
   170→              </div>
   171→              {virtualFields.map((field) => (
   172→                <RecordFieldRow
   173→                  key={field.key}
   174→                  field={field}
   175→                  value={row[field.key]}
   176→                  sectionKey={section.key}
   177→                  rowIndex={clampedRecord}
   178→                  fullWidth
   179→                />
   180→              ))}
   181→            </div>
   182→          )}
   183→
   184→          {/* Main 2-column field grid */}
   185→          {fieldPairs.length > 0 && (
   186→            <div className="grid grid-cols-2 gap-x-6 gap-y-3">
   187→              {fieldPairs.map((pair, pairIdx) => (
   188→                <React.Fragment key={pairIdx}>
   189→                  <RecordFieldRow
   190→                    field={pair[0]!}
   191→                    value={row[pair[0]!.key]}
   192→                    sectionKey={section.key}
   193→                    rowIndex={clampedRecord}
   194→                    fullWidth={false}
   195→                  />
   196→                  {pair[1] ? (
   197→                    <RecordFieldRow
   198→                      field={pair[1]}
   199→                      value={row[pair[1].key]}
   200→                      sectionKey={section.key}
   201→                      rowIndex={clampedRecord}
   202→                      fullWidth={false}
   203→                    />
   204→                  ) : null}
   205→                </React.Fragment>
   206→              ))}
   207→            </div>
   208→          )}
   209→
   210→          {/* Textarea fields — full width */}
   211→          {textareaFields.length > 0 && (
   212→            <div className="space-y-3">
   213→              {textareaFields.map((field) => (
   214→                <RecordFieldRow
   215→                  key={field.key}
   216→                  field={field}
   217→                  value={row[field.key]}
   218→                  sectionKey={section.key}
   219→                  rowIndex={clampedRecord}
   220→                  fullWidth
   221→                />
   222→              ))}
   223→            </div>
   224→          )}
   225→
   226→          {/* Array fields — full width */}
   227→          {arrayFields.length > 0 && (
   228→            <div className="space-y-3">
   229→              {arrayFields.map((field) => (
   230→                <RecordFieldRow
   231→                  key={field.key}
   232→                  field={field}
   233→                  value={row[field.key]}
   234→                  sectionKey={section.key}
   235→                  rowIndex={clampedRecord}
   236→                  fullWidth
   237→                />
   238→              ))}
   239→            </div>
   240→          )}
   241→
   242→          {/* Object fields (nested arrays/objects) — full width */}
   243→          {objectFields.length > 0 && (
   244→            <div className="space-y-4">
   245→              {objectFields.map((field) => (
   246→                <RecordFieldRow
   247→                  key={field.key}
   248→                  field={field}
   249→                  value={row[field.key]}
   250→                  sectionKey={section.key}
   251→                  rowIndex={clampedRecord}
   252→                  fullWidth
   253→                />
   254→              ))}
   255→            </div>
   256→          )}
   257→
   258→          {/* Read-only fields (qdrant tags, etc.) — collapsed */}
   259→          {readOnlyFields.length > 0 && (
   260→            <ReadonlyFieldsBlock fields={readOnlyFields} row={row} />
   261→          )}
   262→        </div>
   263→      )}
   264→    </div>
   265→  );
   266→}

   280→function RecordFieldRow({
   281→  field,
   282→  value,
   283→  sectionKey,
   284→  rowIndex,
   285→  fullWidth,
   286→}: RecordFieldRowProps) {
   287→  const updateCell = useStore((s) => s.updateCell);
   288→  const updateNestedCell = useStore((s) => s.updateNestedCell);
   289→  const addNestedRow = useStore((s) => s.addNestedRow);
   290→  const removeNestedRow = useStore((s) => s.removeNestedRow);
   291→
   292→  const isAutoFilled = field.autoFilled;
   293→
   294→  const labelContent = (
   295→    <div className="flex items-center gap-1.5">
   296→      <Label
   297→        className={`text-xs font-medium ${isAutoFilled ? 'text-primary' : 'text-foreground/80'}`}
   298→      >
   299→        {field.label}
   300→      </Label>
   301→      {isAutoFilled && (
   302→        <Badge
   303→          variant="outline"
   304→          className="text-[9px] px-1 py-0 h-3.5 text-primary/70 border-primary/30"
   305→        >
   306→          авто
   307→        </Badge>
   308→      )}
   309→      {field.hint && (
   310→        <Tooltip>
   311→          <TooltipTrigger asChild>
   312→            <span className="text-muted-foreground/50 cursor-help text-[10px] leading-none">?</span>
   313→          </TooltipTrigger>
   314→          <TooltipContent side="top" sideOffset={4} className="max-w-xs">
   315→            <p className="text-xs">{field.hint}</p>
   316→          </TooltipContent>
   317→        </Tooltip>
   318→      )}
   319→    </div>
   320→  );
   321→
   322→  const inputEl = (
   323→    <FieldInput
   324→      field={field}
   325→      value={value}
   326→      sectionKey={sectionKey}
   327→      rowIndex={rowIndex}
   328→      updateCell={updateCell}
   329→      updateNestedCell={updateNestedCell}
   330→      addNestedRow={addNestedRow}
   331→      removeNestedRow={removeNestedRow}
   332→    />
   333→  );
   334→
   335→  // For object fields (nested tables), render differently
   336→  if (field.type === 'object' && field.nestedFields) {
   337→    return (
   338→      <div className={fullWidth ? '' : 'col-span-2'}>
   339→        <div className="flex items-center gap-2 mb-2">{labelContent}</div>
   340→        <NestedFieldBlock
   341→          field={field}
   342→          value={value}
   343→          sectionKey={sectionKey}
   344→          rowIndex={rowIndex}
   345→          updateNestedCell={updateNestedCell}
   346→          addNestedRow={addNestedRow}
   347→          removeNestedRow={removeNestedRow}
   348→        />
   349→      </div>
   350→    );
   351→  }
   352→
   353→  // For array fields that need more space
   354→  if (field.type === 'array' && fullWidth) {
   355→    return (
   356→      <div>
   357→        {labelContent}
   358→        <div className="mt-1">{inputEl}</div>
   359→      </div>
   360→    );
   361→  }
   362→
   363→  // For textarea in grid, span 2 columns
   364→  if (field.type === 'textarea') {
   365→    return (
   366→      <div className="col-span-2">
   367→        {labelContent}
   368→        <div className="mt-1">{inputEl}</div>
   369→      </div>
   370→    );
   371→  }
   372→
   373→  return (
   374→    <div>
   375→      {labelContent}
   376→      <div className="mt-1">{inputEl}</div>
   377→    </div>
   378→  );
   379→}

   381→// =============================================================================
   382→// FieldInput — renders the actual input control for a field
   383→// =============================================================================
   384→
   385→interface FieldInputProps {
   386→  field: FieldDef;
   387→  value: unknown;
   388→  sectionKey: string;
   389→  rowIndex: number;
   390→  updateCell: (sectionKey: string, rowIndex: number, fieldKey: string, value: unknown) => void;
   391→  updateNestedCell: (sectionKey: string, rowIndex: number, fieldKey: string, nestedIndex: number, nestedFieldKey: string, value: unknown) => void;
   392→  addNestedRow: (sectionKey: string, rowIndex: number, fieldKey: string) => void;
   393→  removeNestedRow: (sectionKey: string, rowIndex: number, fieldKey: string, nestedIndex: number) => void;
   394→}
   395→
   396→function FieldInput({
   397→  field,
   398→  value,
   399→  sectionKey,
   400→  rowIndex,
   401→  updateCell,
   402→}: FieldInputProps) {
   403→  const isAutoFilled = field.autoFilled;
   404→  const baseInputClass = isAutoFilled
   405→    ? 'h-9 text-sm border-primary/25 bg-primary/5 focus-visible:ring-primary/40'
   406→    : 'h-9 text-sm focus-visible:ring-1 focus-visible:ring-ring';
   407→
   408→  switch (field.type) {
   409→    case 'text':
   410→      return (
   411→        <Input
   412→          type="text"
   413→          value={(value as string) || ''}
   414→          placeholder={field.placeholder}
   415→          className={baseInputClass}
   416→          onChange={(e) => updateCell(sectionKey, rowIndex, field.key, e.target.value)}
   417→        />
   418→      );
   419→
   420→    case 'textarea':
   421→      return (
   422→        <Textarea
   423→          value={(value as string) || ''}
   424→          placeholder={field.placeholder}
   425→          rows={3}
   426→          className="min-h-[80px] text-sm resize-y focus-visible:ring-1 focus-visible:ring-ring"
   427→          onChange={(e) => updateCell(sectionKey, rowIndex, field.key, e.target.value)}
   428→        />
   429→      );
   430→
   431→    case 'number':
   432→      return (
   433→        <Input
   434→          type="number"
   435→          step="any"
   436→          value={value === null || value === undefined ? '' : String(value)}
   437→          placeholder={field.placeholder}
   438→          className={baseInputClass}
   439→          onChange={(e) => {
   440→            const v = e.target.value;
   441→            updateCell(sectionKey, rowIndex, field.key, v === '' ? null : Number(v));
   442→          }}
   443→        />
   444→      );
   445→
   446→    case 'date':
   447→      return (
   448→        <Input
   449→          type="date"
   450→          value={(value as string) || ''}
   451→          className={baseInputClass}
   452→          onChange={(e) => updateCell(sectionKey, rowIndex, field.key, e.target.value || null)}
   453→        />
   454→      );
   455→
   456→    case 'boolean':
   457→      return (
   458→        <div className="flex items-center h-9">
   459→          <Switch
   460→            checked={!!value}
   461→            onCheckedChange={(checked) => updateCell(sectionKey, rowIndex, field.key, checked)}
   462→          />
   463→        </div>
   464→      );
   465→
   466→    case 'select':
   467→      return (
   468→        <Select
   469→          value={(value as string) || ''}
   470→          onValueChange={(v) => updateCell(sectionKey, rowIndex, field.key, v)}
   471→        >
   472→          <SelectTrigger className={isAutoFilled ? 'h-9 text-sm border-primary/25 bg-primary/5 focus:ring-primary/40 w-full' : 'h-9 text-sm w-full'}>
   473→            <SelectValue placeholder={field.placeholder || 'Выберите…'} />
   474→          </SelectTrigger>
   475→          <SelectContent>
   476→            {(field.options || []).map((opt) => (
   477→              <SelectItem key={opt} value={opt}>
   478→                {opt}
   479→              </SelectItem>
   480→            ))}
   481→          </SelectContent>
   482→        </Select>
   483→      );
   484→
   485→    case 'array': {
   486→      const arr = Array.isArray(value) ? (value as string[]) : [];
   487→      return (
   488→        <Input
   489→          type="text"
   490→          value={arr.join(', ')}
   491→          placeholder="через запятую"
   492→          className={baseInputClass}
   493→          onChange={(e) => {
   494→            const items = e.target.value
   495→              .split(',')
   496→              .map((s) => s.trim())
   497→              .filter(Boolean);
   498→            updateCell(sectionKey, rowIndex, field.key, items);
   499→          }}
   500→        />
   501→      );
   502→    }
   503→
   504→    case 'ref':
   505→      return <RefSelect field={field} value={value} sectionKey={sectionKey} rowIndex={rowIndex} />;
   506→
   507→    default:
   508→      return <span className="text-muted-foreground text-xs">—</span>;
   509→  }
   510→}

   512→// =============================================================================
   513→// RefSelect — dropdown populated from another section's data
   514→// =============================================================================
   515→
   516→function RefSelect({
   517→  field,
   518→  value,
   519→  sectionKey,
   520→  rowIndex,
   521→}: {
   522→  field: FieldDef;
   523→  value: unknown;
   524→  sectionKey: string;
   525→  rowIndex: number;
   526→}) {
   527→  const updateCell = useStore((s) => s.updateCell);
   528→  const allData = useStore((s) => s.data);
   529→  const refSection = field.refSection;
   530→  const refLabelField = field.refLabelField;
   531→
   532→  const options = useMemo(() => {
   533→    if (!refSection) return [];
   534→    const rows = (allData[refSection] || []) as Record<string, unknown>[];
   535→    return rows
   536→      .map((row, idx) => {
   537→        const label = refLabelField
   538→          ? String(row[refLabelField] || '')
   539→          : `Запись ${idx + 1}`;
   540→        return { idx, label };
   541→      })
   542→      .filter((o) => o.label);
   543→  }, [allData, refSection, refLabelField]);
   544→
   545→  const currentIdx = value as number | null;
   546→  const currentValue =
   547→    currentIdx !== null && currentIdx !== undefined ? String(currentIdx) : '';
   548→  const currentLabel =
   549→    currentIdx !== null && currentIdx !== undefined
   550→      ? options.find((o) => o.idx === currentIdx)?.label || ''
   551→      : '';
   552→
   553→  const isVirtual = field.virtual;
   554→
   555→  return (
   556→    <Select
   557→      value={currentValue}
   558→      onValueChange={(v) => updateCell(sectionKey, rowIndex, field.key, Number(v))}
   559→    >
   560→      <SelectTrigger
   561→        className={`h-9 text-sm w-full ${
   562→          isVirtual
   563→            ? 'border-primary/30 bg-primary/5 focus:ring-1 focus:ring-primary'
   564→            : ''
   565→        }`}
   566→      >
   567→        <SelectValue placeholder={field.placeholder || 'Выберите…'}>
   568→          {currentLabel || (field.placeholder || 'Выберите…')}
   569→        </SelectValue>
   570→      </SelectTrigger>
   571→      <SelectContent>
   572→        {options.length === 0 && (
   573→          <SelectItem value="_empty" disabled>
   574→            Нет записей в справочнике
   575→          </SelectItem>
   576→        )}
   577→        {options.map((opt) => (
   578→          <SelectItem key={opt.idx} value={String(opt.idx)}>
   579→            {opt.label}
   580→          </SelectItem>
   581→        ))}
   582→      </SelectContent>
   583→    </Select>
   584→  );
   585→}

   587→// =============================================================================
   588→// NestedFieldBlock — for object fields with nested sub-fields
   589→// =============================================================================
   590→
   591→function NestedFieldBlock({
   592→  field,
   593→  value,
   594→  sectionKey,
   595→  rowIndex,
   596→  updateNestedCell,
   597→  addNestedRow,
   598→  removeNestedRow,
   599→}: {
   600→  field: FieldDef;
   601→  value: unknown;
   602→  sectionKey: string;
   603→  rowIndex: number;
   604→  updateNestedCell: (sectionKey: string, rowIndex: number, fieldKey: string, nestedIndex: number, nestedFieldKey: string, value: unknown) => void;
   605→  addNestedRow: (sectionKey: string, rowIndex: number, fieldKey: string) => void;
   606→  removeNestedRow: (sectionKey: string, rowIndex: number, fieldKey: string, nestedIndex: number) => void;
   607→}) {
   608→  const [open, setOpen] = useState(false);
   609→  const editableNestedFields = (field.nestedFields || []).filter(
   610→    (nf) => !nf.readOnly,
   611→  );
   612→
   613→  // Array of nested objects (e.g., points, scenarios, cargo lists)
   614→  if (Array.isArray(value)) {
   615→    const items = value as Record<string, unknown>[];
   616→    return (
   617→      <Collapsible open={open} onOpenChange={setOpen}>
   618→        <CollapsibleTrigger asChild>
   619→          <button className="flex items-center gap-2 text-xs text-muted-foreground hover:text-foreground transition-colors">
   620→            {open ? (
   621→              <ChevronDown className="size-3.5" />
   622→            ) : (
   623→              <ChevronRight className="size-3.5" />
   624→            )}
   625→            <Badge variant="secondary" className="text-xs">
   626→              {items.length}{' '}
   627→              {items.length === 1
   628→                ? 'запись'
   629→                : items.length < 5
   630→                  ? 'записи'
   631→                  : 'записей'}
   632→            </Badge>
   633→            <span className="underline">развернуть</span>
   634→          </button>
   635→        </CollapsibleTrigger>
   636→        <CollapsibleContent>
   637→          <div className="mt-2 border rounded-lg overflow-hidden">
   638→            {/* Header */}
   639→            <div className="grid bg-muted/50 border-b px-3 py-2 text-xs font-medium text-muted-foreground">
   640→              <div className="flex gap-4">
   641→                <span className="w-8 text-center">#</span>
   642→                {editableNestedFields.map((nf) => (
   643→                  <span
   644→                    key={nf.key}
   645→                    className="min-w-[120px]"
   646→                  >
   647→                    {nf.label}
   648→                  </span>
   649→                ))}
   650→                <span className="w-7" />
   651→              </div>
   652→            </div>
   653→            {/* Rows */}
   654→            {items.map((item, ni) => (
   655→              <div
   656→                key={ni}
   657→                className="grid border-b last:border-b-0 hover:bg-muted/20 px-3 py-1.5"
   658→              >
   659→                <div className="flex gap-4 items-center">
   660→                  <span className="w-8 text-center text-xs text-muted-foreground">
   661→                    {ni + 1}
   662→                  </span>
   663→                  {editableNestedFields.map((nf) => (
   664→                    <div key={nf.key} className="min-w-[120px]">
   665→                      <NestedFieldInput
   666→                        nestedField={nf}
   667→                        value={item[nf.key]}
   668→                        onChange={(v) =>
   669→                          updateNestedCell(
   670→                            sectionKey,
   671→                            rowIndex,
   672→                            field.key,
   673→                            ni,
   674→                            nf.key,
   675→                            v,
   676→                          )
   677→                        }
   678→                      />
   679→                    </div>
   680→                  ))}
   681→                  <div className="w-7">
   682→                    <Button
   683→                      variant="ghost"
   684→                      size="icon"
   685→                      className="size-6 text-muted-foreground hover:text-destructive"
   686→                      onClick={() =>
   687→                        removeNestedRow(sectionKey, rowIndex, field.key, ni)
   688→                      }
   689→                    >
   690→                      <Minus className="size-3" />
   691→                    </Button>
   692→                  </div>
   693→                </div>
   694→              </div>
   695→            ))}
   696→            {/* Add row */}
   697→            <div className="px-3 py-1.5 border-t">
   698→              <Button
   699→                variant="ghost"
   700→                size="sm"
   701→                className="h-7 text-xs"
   702→                onClick={() => addNestedRow(sectionKey, rowIndex, field.key)}
   703→              >
   704→                <Plus className="mr-1 size-3" />
   705→                Добавить
   706→              </Button>
   707→            </div>
   708→          </div>
   709→        </CollapsibleContent>
   710→      </Collapsible>
   711→    );
   712→  }
   713→
   714→  // Single nested object (e.g., ztb_boundaries)
   715→  if (field.nestedFields && typeof value === 'object' && value !== null) {
   716→    const obj = value as Record<string, unknown>;
   717→    return (
   718→      <div className="grid grid-cols-2 gap-x-6 gap-y-3">
   719→        {editableNestedFields.map((nf) => (
   720→          <div key={nf.key}>
   721→            <Label className="text-xs font-medium text-foreground/80">
   722→              {nf.label}
   723→            </Label>
   724→            <div className="mt-1">
   725→              <NestedFieldInput
   726→                nestedField={nf}
   727→                value={obj[nf.key]}
   728→                onChange={(v) => {
   729→                  const updated = { ...obj, [nf.key]: v };
   730→                  useStore
   731→                    .getState()
   732→                    .updateCell(sectionKey, rowIndex, field.key, updated);
   733→                }}
   734→              />
   735→            </div>
   736→          </div>
   737→        ))}
   738→      </div>
   739→    );
   740→  }
   741→
   742→  return null;
   743→}

   745→// =============================================================================
   746→// NestedFieldInput — input for a single nested field
   747→// =============================================================================
   748→
   749→function NestedFieldInput({
   750→  nestedField,
   751→  value,
   752→  onChange,
   753→}: {
   754→  nestedField: FieldDef;
   755→  value: unknown;
   756→  onChange: (value: unknown) => void;
   757→}) {
   758→  const v = value ?? '';
   759→
   760→  switch (nestedField.type) {
   761→    case 'boolean':
   762→      return (
   763→        <Switch
   764→          checked={!!value}
   765→          onCheckedChange={(checked) => onChange(checked)}
   766→          className="scale-75"
   767→        />
   768→      );
   769→    case 'number':
   770→      return (
   771→        <input
   772→          type="number"
   773→          step="any"
   774→          value={v === '' || v === null ? '' : String(v)}
   775→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
   776→          onChange={(e) => {
   777→            const nv = e.target.value;
   778→            onChange(nv === '' ? null : Number(nv));
   779→          }}
   780→        />
   781→      );
   782→    case 'select':
   783→      return (
   784→        <select
   785→          value={String(v)}
   786→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
   787→          onChange={(e) => onChange(e.target.value)}
   788→        >
   789→          <option value="">—</option>
   790→          {(nestedField.options || []).map((opt) => (
   791→            <option key={opt} value={opt}>
   792→              {opt}
   793→            </option>
   794→          ))}
   795→        </select>
   796→      );
   797→    default:
   798→      return (
   799→        <input
   800→          type="text"
   801→          value={String(v)}
   802→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
   803→          onChange={(e) => onChange(e.target.value)}
   804→        />
   805→      );
   806→  }
   807→}

   812→
   813→function ReadonlyFieldsBlock({
   814→  fields,
   815→  row,
   816→}: {
   817→  fields: FieldDef[];
   818→  row: Record<string, unknown>;
   819→}) {
   820→  const [open, setOpen] = useState(false);
   821→
   822→  const hasValues = fields.some(
   823→    (f) => row[f.key] !== '' && row[f.key] !== null && row[f.key] !== undefined,
   824→  );
   825→
   826→  // If no values and closed, show minimal indicator
   827→  if (!hasValues && !open) {
   828→    return null;
   829→  }
   830→
   831→  return (
   832→    <Collapsible open={open} onOpenChange={setOpen}>
   833→      <CollapsibleTrigger asChild>
   834→        <button className="flex items-center gap-2 text-xs text-muted-foreground hover:text-foreground transition-colors">
   835→          {open ? (
   836→            <ChevronDown className="size-3.5" />
   837→          ) : (
   838→            <ChevronRight className="size-3.5" />
   839→          )}
   840→          <span className="font-medium">Системные теги и метаданные</span>
   841→          <Badge variant="outline" className="text-[9px] px-1 py-0 h-4">
   842→            {fields.length} полей
   843→          </Badge>
   844→          {hasValues && !open && (
   845→            <Badge variant="secondary" className="text-[9px] px-1 py-0 h-4 bg-primary/10 text-primary border-primary/25">
   846→              есть данные
   847→            </Badge>
   848→          )}
   849→        </button>
   850→      </CollapsibleTrigger>
   851→      <CollapsibleContent>
   852→        <div className="mt-2 grid grid-cols-3 gap-x-6 gap-y-2 p-3 rounded-lg bg-muted/30 border">
   853→          {fields.map((f) => (
   854→            <div key={f.key} className="flex items-center justify-between gap-2">
   855→              <span className="text-xs text-muted-foreground truncate" title={f.label}>
   856→                {f.label}
   857→              </span>
   858→              <span className="text-xs font-mono text-muted-foreground/70 truncate max-w-[160px]">
   859→                {row[f.key] === null
   860→                  ? 'null'
   861→                  : Array.isArray(row[f.key])
   862→                    ? `[${(row[f.key] as unknown[]).length}]`
   863→                    : String(row[f.key]) || '—'}
   864→              </span>
   865→            </div>
   866→          ))}
   867→        </div>
   868→      </CollapsibleContent>
   869→    </Collapsible>
   870→  );
   871→}
