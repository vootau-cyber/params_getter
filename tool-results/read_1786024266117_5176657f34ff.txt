    44→    44→
    45→    45→export type FieldStatus = 'empty' | 'invalid' | 'ok';
    46→    46→
    47→    47→/**
    48→    48→ * Determine if a field value is considered "empty" (unfilled).
    49→    49→ * Booleans, readOnly, virtual, and object fields are always 'ok'.
    50→    50→ */
    51→    51→export function getFieldStatus(field: FieldDef, value: unknown): FieldStatus {
    52→    52→  // Skip types that don't need validation
    53→    53→  if (field.readOnly || field.virtual) return 'ok';
    54→    54→  if (field.type === 'boolean') return 'ok';
    55→    55→  if (field.type === 'object') return 'ok';
    56→    56→
    57→    57→  switch (field.type) {
    58→    58→    case 'text':
    59→    59→    case 'textarea':
    60→    60→      return (value as string) === '' ? 'empty' : 'ok';
    61→    61→    case 'number':
    62→    62→      return value === null || value === undefined ? 'empty' : 'ok';
    63→    63→    case 'date':
    64→    64→      return value === null || value === undefined || value === '' ? 'empty' : 'ok';
    65→    65→    case 'select':
    66→    66→      return (value as string) === '' || value === null ? 'empty' : 'ok';
    67→    67→    case 'array':
    68→    68→      return Array.isArray(value) && value.length === 0 ? 'empty' : 'ok';
    69→    69→    case 'ref':
    70→    70→      return value === null || value === undefined ? 'empty' : 'ok';
    71→    71→    default:
    72→    72→      return 'ok';
    73→    73→  }
    74→    74→}
    75→    75→
    76→    76→/** Count empty + invalid fields in a record */
    77→    77→export function countFieldIssues(
    78→    78→  fields: FieldDef[],
    79→    79→  row: Record<string, unknown>,
    80→    80→): number {
    81→    81→  let count = 0;
    82→    82→  for (const f of fields) {
    83→    83→    const status = getFieldStatus(f, row[f.key]);
    84→    84→    if (status !== 'ok') count++;
    85→    85→  }
    86→    86→  return count;
    87→    87→}

   491→   491→function FieldInput({
   492→   492→  field,
   493→   493→  value,
   494→   494→  sectionKey,
   495→   495→  rowIndex,
   496→   496→  status,
   497→   497→  updateCell,
   498→   498→}: FieldInputProps) {
   499→   499→  const isAutoFilled = field.autoFilled;
   500→   500→  const isEmpty = status === 'empty';
   501→   501→
   502→   502→  // Build class based on status
   503→   503→  const baseInputClass = isAutoFilled
   504→   504→    ? 'h-9 text-sm border-primary/25 bg-primary/5 focus-visible:ring-primary/40'
   505→   505→    : isEmpty
   506→   506→      ? 'h-9 text-sm border-destructive/50 bg-destructive/5 focus-visible:ring-destructive/40'
   507→   507→      : 'h-9 text-sm focus-visible:ring-1 focus-visible:ring-ring';
   508→   508→
   509→   509→  switch (field.type) {
   510→   510→    case 'text':
   511→   511→      return (
   512→   512→        <Input
   513→   513→          type="text"
   514→   514→          value={(value as string) || ''}
   515→   515→          placeholder={field.placeholder}
   516→   516→          className={baseInputClass}
   517→   517→          onChange={(e) =>
   518→   518→            updateCell(sectionKey, rowIndex, field.key, e.target.value)
   519→   519→          }
   520→   520→        />
   521→   521→      );
   522→   522→
   523→   523→    case 'textarea':
   524→   524→      return (
   525→   525→        <Textarea
   526→   526→          value={(value as string) || ''}

   655→   655→function RefSelect({
   656→   656→  field,
   657→   657→  value,
   658→   658→  sectionKey,
   659→   659→  rowIndex,
   660→   660→  isEmpty,
   661→   661→}: {
   662→   662→  field: FieldDef;
   663→   663→  value: unknown;
   664→   664→  sectionKey: string;
   665→   665→  rowIndex: number;
   666→   666→  isEmpty: boolean;
   667→   667→}) {
   668→   668→  const updateCell = useStore((s) => s.updateCell);
   669→   669→  const allData = useStore((s) => s.data);
   670→   670→  const refSection = field.refSection;
   671→   671→  const refLabelField = field.refLabelField;
   672→   672→
   673→   673→  const options = useMemo(() => {
   674→   674→    if (!refSection) return [];
   675→   675→    const rows = (allData[refSection] || []) as Record<string, unknown>[];
   676→   676→    return rows
   677→   677→      .map((row, idx) => {
   678→   678→        const label = refLabelField
   679→   679→          ? String(row[refLabelField] || '')
   680→   680→          : `Запись ${idx + 1}`;
   681→   681→        return { idx, label };
   682→   682→      })
   683→   683→      .filter((o) => o.label);
   684→   684→  }, [allData, refSection, refLabelField]);
   685→   685→
   686→   686→  const currentIdx = value as number | null;
   687→   687→  const currentValue =
   688→   688→    currentIdx !== null && currentIdx !== undefined
   689→   689→      ? String(currentIdx)
   690→   690→      : '';
   691→   691→  const currentLabel =
   692→   692→    currentIdx !== null && currentIdx !== undefined
   693→   693→      ? options.find((o) => o.idx === currentIdx)?.label || ''
   694→   694→      : '';
   695→   695→
   696→   696→  const isVirtual = field.virtual;
   697→   697→  const triggerClass = isEmpty
   698→   698→    ? 'h-9 text-sm w-full border-destructive/50 bg-destructive/5 focus:ring-destructive/40'
   699→   699→    : isVirtual
   700→   700→      ? 'h-9 text-sm w-full border-primary/30 bg-primary/5 focus:ring-1 focus:ring-primary'
   701→   701→      : 'h-9 text-sm w-full';
   702→   702→
   703→   703→  return (
   704→   704→    <Select
   705→   705→      value={currentValue}
   706→   706→      onValueChange={(v) =>
   707→   707→        updateCell(sectionKey, rowIndex, field.key, Number(v))
   708→   708→      }
   709→   709→    >
   710→   710→      <SelectTrigger className={triggerClass}>
   711→   711→        <SelectValue placeholder={field.placeholder || 'Выберите…'}>
   712→   712→          {currentLabel || (field.placeholder || 'Выберите…')}
   713→   713→        </SelectValue>
   714→   714→      </SelectTrigger>
   715→   715→      <SelectContent>
   716→   716→        {options.length === 0 && (
   717→   717→          <SelectItem value="_empty" disabled>
   718→   718→            Нет записей в справочнике
   719→   719→          </SelectItem>
   720→   720→        )}
   721→   721→        {options.map((opt) => (
   722→   722→          <SelectItem key={opt.idx} value={String(opt.idx)}>
   723→   723→            {opt.label}
   724→   724→          </SelectItem>
   725→   725→        ))}
   726→   726→      </SelectContent>
   727→   727→    </Select>
   728→   728→  );
   729→   729→}

   913→   913→function NestedFieldInput({
   914→   914→  nestedField,
   915→   915→  value,
   916→   916→  onChange,
   917→   917→}: {
   918→   918→  nestedField: FieldDef;
   919→   919→  value: unknown;
   920→   920→  onChange: (value: unknown) => void;
   921→   921→}) {
   922→   922→  const v = value ?? '';
   923→   923→
   924→   924→  switch (nestedField.type) {
   925→   925→    case 'boolean':
   926→   926→      return (
   927→   927→        <Switch
   928→   928→          checked={!!value}
   929→   929→          onCheckedChange={(checked) => onChange(checked)}
   930→   930→          className="scale-75"
   931→   931→        />
   932→   932→      );
   933→   933→    case 'number':
   934→   934→      return (
   935→   935→        <input
   936→   936→          type="number"
   937→   937→          step="any"
   938→   938→          value={v === '' || v === null ? '' : String(v)}
   939→   939→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
   940→   940→          onChange={(e) => {
   941→   941→            const nv = e.target.value;
   942→   942→            onChange(nv === '' ? null : Number(nv));
   943→   943→          }}
   944→   944→        />
   945→   945→      );
   946→   946→    case 'select':
   947→   947→      return (
   948→   948→        <select
   949→   949→          value={String(v)}
   950→   950→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
   951→   951→          onChange={(e) => onChange(e.target.value)}
   952→   952→        >
   953→   953→          <option value="">—</option>
   954→   954→          {(nestedField.options || []).map((opt) => (
   955→   955→            <option key={opt} value={opt}>
   956→   956→              {opt}
   957→   957→            </option>
   958→   958→          ))}
   959→   959→        </select>
   960→   960→      );
   961→   961→    default:
   962→   962→      return (
   963→   963→        <input
   964→   964→          type="text"
   965→   965→          value={String(v)}
   966→   966→          className="w-full h-7 text-xs border rounded-md px-2 bg-background focus:outline-none focus:ring-1 focus:ring-ring"
   967→   967→          onChange={(e) => onChange(e.target.value)}
   968→   968→        />
   969→   969→      );
   970→   970→  }
   971→   971→}
